# A view that identifies any records in MAD_ADDRESS_VARIANTS that have a COMMUNITY_ID value that doesn't
# match the COMMUNITY_ID in MAD_MASTER_ADDRESS, but should.  If the MASTER_ADDRESS_ID assigned to the address
# variant record is correct, then the COMMUNITY_ID should be updated to reflect the one in MAD_MASTER_ADDRESS
# or vice versa.
# DM - 2/8/17
# Updated 3/9/17 to focus on street names that match.


CREATE OR REPLACE VIEW MADV_QA_AV_COMM_MISMATCH AS
SELECT AV.MASTER_ADDRESS_ID, AV.INPUT_FULL_ADDRESS, MA.ADDRESS_POINT_ID, AV.COMMUNITY_ID AV_COMM_ID, MA.COMMUNITY_ID MA_COMM_ID, 
AV.STREET_NAME_ID AV_STREET_NAME_ID, MSN1.STREET_NAME AV_STREETNAME, MSN1.COMMUNITY_ID AV_MSN_COMM_ID,
MA.STREET_NAME_ID MA_STREET_NAME_ID, MSN2.STREET_NAME MA_STREETNAME, MSN2.COMMUNITY_ID MA_MSN_COMM_ID
FROM MAD_ADDRESS_VARIANTS AV, MAD_MASTER_ADDRESS MA, MAD_MASTER_STREET_NAME MSN1, MAD_MASTER_STREET_NAME MSN2
WHERE MA.MASTER_ADDRESS_ID = AV.MASTER_ADDRESS_ID
AND MSN1.STREET_NAME_ID = AV.STREET_NAME_ID
AND MSN2.STREET_NAME_ID = MA.STREET_NAME_ID
AND MSN1.STREET_NAME = MSN2.STREET_NAME
AND AV.COMMUNITY_ID <> MA.COMMUNITY_ID
ORDER BY AV.COMMUNITY_ID, AV.INPUT_FULL_STREET_NAME, AV.MASTER_ADDRESS_ID;
