# A view that identifies any records in MAD_ADDRESS_VARIANTS that have a GEOGRAPHIC_TOWN_ID value that doesn't
# match the GEOGRAPHIC_TOWN_ID in MAD_MASTER_ADDRESS, but should.  If the MASTER_ADDRESS_ID assigned to the address
# variant record is correct, then the GEOGRAPHIC_TOWN_ID should be updated to reflect the one in MAD_MASTER_ADDRESS
# or vice versa.
# DM - 2/8/17


CREATE OR REPLACE VIEW MADV_QA_AV_TOWN_MISMATCH AS
SELECT AV.MASTER_ADDRESS_ID, AV.INPUT_FULL_ADDRESS, MA.ADDRESS_POINT_ID, AV.GEOGRAPHIC_TOWN_ID AV_GEO_TOWN_ID, MA.GEOGRAPHIC_TOWN_ID MA_GEO_TOWN_ID 
FROM MAD_ADDRESS_VARIANTS AV, MAD_MASTER_ADDRESS MA
WHERE MA.MASTER_ADDRESS_ID = AV.MASTER_ADDRESS_ID
AND MA.GEOGRAPHIC_TOWN_ID <> AV.GEOGRAPHIC_TOWN_ID
ORDER BY AV.GEOGRAPHIC_TOWN_ID, AV.INPUT_FULL_STREET_NAME, AV.MASTER_ADDRESS_ID;
