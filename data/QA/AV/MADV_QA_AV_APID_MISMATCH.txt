# A view that identifies any records in MAD_ADDRESS_VARIANTS that have an ADDRESS_POINT_ID value that doesn't
# match the ADDRESS_POINT_ID in MAD_MASTER_ADDRESS, but should.  If the MASTER_ADDRESS_ID assigned to the address
# variant record is correct, then the ADDRESS_POINT_ID should be updated to reflect the one in MAD_MASTER_ADDRESS.
# DM - 1/17/17


CREATE OR REPLACE VIEW MADV_QA_AV_APID_MISMATCH AS
SELECT AV.INPUT_FULL_ADDRESS, AV.SOURCE_NAME_ID, AV.GEOGRAPHIC_TOWN_ID, AV.MASTER_ADDRESS_ID, AV.ADDRESS_POINT_ID AV_APID, MA.ADDRESS_POINT_ID MA_APID, MA.ADDRESS_STATUS
FROM MAD_ADDRESS_VARIANTS AV, mad_master_address ma, mad_address_pointm ap
WHERE AV.MASTER_ADDRESS_ID = MA.MASTER_ADDRESS_ID
and ma.address_point_id = ap.address_point_id
AND MA.ADDRESS_POINT_ID IS NOT NULL 
AND MA.ADDRESS_POINT_ID NOT LIKE 'MULTI%'
AND MA.ADDRESS_STATUS IN ('ACTIVE', 'LINKED')
AND (AV.ADDRESS_POINT_ID IS NULL 
     OR (AV.ADDRESS_POINT_ID IS NOT NULL AND MA.ADDRESS_POINT_ID <> AV.ADDRESS_POINT_ID AND MA.FULL_NUMBER_STANDARDIZED <> '0' ))
and av.loc_id = ap.loc_id
ORDER BY AV.GEOGRAPHIC_TOWN_ID, MA.FULL_NUMBER_STANDARDIZED, MA.STREET_NAME_ID;
