# View identifies any records in MAD_ADDRESS_VARIANTS that have an invalid value in DATE_CREATED.
# DM - 6/25/19

CREATE OR REPLACE VIEW MADV_QA_AV_DATE_CREATED AS
SELECT * FROM MAD_ADDRESS_VARIANTS AV
WHERE TO_NUMBER(SUBSTR(DATE_CREATED, 0, 4)) NOT BETWEEN 1900 AND (SELECT TO_NUMBER(TO_CHAR(sysdate, 'YYYY')) FROM DUAL)
OR TO_NUMBER(SUBSTR(DATE_CREATED, 5, 2)) NOT BETWEEN 01 AND 12
OR TO_NUMBER(SUBSTR(DATE_CREATED, 7, 2)) NOT BETWEEN 01 AND 31
OR NOT REGEXP_LIKE(DATE_CREATED, '^[0-9]{8}$')
OR DATE_CREATED IS NULL;
