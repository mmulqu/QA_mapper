# View identifies any records in MAD_ADDRESS_VARIANTS with an unacceptable UNIT_TYPE value.  
# "Unacceptable" means it is a non-null value not within the acceptable set of domain values.
# DM - 6/24/19

CREATE OR REPLACE VIEW MADV_QA_AV_DOM_UNITTYPE AS
SELECT * FROM MAD_ADDRESS_VARIANTS AV
WHERE UNIT_TYPE IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'UNIT_TYPE'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_ADDRESS_VARIANTS'
AND AV.UNIT_TYPE = DOM.DOMAIN_VALUE)
ORDER BY UNIT_TYPE, GEOGRAPHIC_TOWN_ID, STREET_NAME_ID, FULL_NUMBER_STANDARDIZED, UNIT;
