# A view that identifies any records in MAD_ADDRESS_VARIANTS that have a SITE_NAME_ID and/or SITE_ID value that doesn't
# match the SITE_NAME_ID and/or SITE_ID in MAD_MASTER_ADDRESS, but should.  If the MASTER_ADDRESS_ID assigned to the address
# variant record is correct, then the SITE_NAME_ID and/or SITE_ID should be updated to reflect the one in MAD_MASTER_ADDRESS
# or vice versa.
# DM - 2/7/17
# Added 'AND MA.ADDRESS_POINT_ID NOT LIKE 'MULTI%'' - DM 6/9/22


CREATE OR REPLACE VIEW MADV_QA_AV_SITE_MISMATCH AS
SELECT AV.MASTER_ADDRESS_ID, AV.INPUT_FULL_ADDRESS, AV.ADDRESS_POINT_ID, AV.GEOGRAPHIC_TOWN_ID, AV.SITE_NAME_ID AV_SITENAMEID, MA2.SITE_NAME_ID MA_SITENAMEID, 
AV.SITE_ID AV_SITE_ID, MA2.SITE_ID MA_SITE_ID
FROM MAD_ADDRESS_VARIANTS AV, MAD_MASTER_ADDRESS MA2
WHERE AV.MASTER_ADDRESS_ID = MA2.MASTER_ADDRESS_ID
AND EXISTS (SELECT * FROM MAD_MASTER_ADDRESS MA
WHERE AV.MASTER_ADDRESS_ID = MA.MASTER_ADDRESS_ID
AND MA.ADDRESS_POINT_ID NOT LIKE 'MULTI%'
AND (((MA.SITE_ID IS NOT NULL AND AV.SITE_ID IS NULL) OR (MA.SITE_ID IS NULL AND AV.SITE_ID IS NOT NULL) OR AV.SITE_ID <> MA.SITE_ID)
OR ((MA.SITE_NAME_ID IS NOT NULL AND AV.SITE_NAME_ID IS NULL) OR (MA.SITE_NAME_ID IS NULL AND AV.SITE_NAME_ID IS NOT NULL) OR AV.SITE_NAME_ID <> MA.SITE_NAME_ID)))
ORDER BY AV.GEOGRAPHIC_TOWN_ID, AV.INPUT_FULL_STREET_NAME, AV.MASTER_ADDRESS_ID;
