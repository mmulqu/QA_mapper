# View identifies any segments in MAD_BASE_STREET_ARC that have a NULL or otherwise
# inappropriate value in the FUNCTIONAL_CLASS field.
# DM - 9/26/16

CREATE OR REPLACE VIEW MADV_QA_BSA_DOM_FUNC_CLASS AS 
SELECT * FROM MAD_BASE_STREET_ARC BSA
WHERE FUNCTIONAL_CLASS IS NULL
OR FUNCTIONAL_CLASS = ''
OR NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'FUNCTIONAL_CLASS'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_BASE_STREET_ARC'
AND BSA.FUNCTIONAL_CLASS = DOM.DOMAIN_VALUE)
ORDER BY FUNCTIONAL_CLASS, COMMUNITY_ID_LEFT_ADDRESS, BASE_SEGMENT_ID;
