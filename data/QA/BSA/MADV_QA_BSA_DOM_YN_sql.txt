# View identifies any segments in MAD_BASE_STREET_ARC where a field requiring a domain value of
# 'Y' or 'N' doesn't have either of those values.  The fields with this domain requirement are:
# DISJOINT_GEOGRAPHY, DISJOINT_RANGE, PAVED, PRIVATE, RAMP, BRIDGE, TOLLWAY, TUNNEL, and FLIPPED.
# DM - 9/27/16

CREATE OR REPLACE VIEW MADV_QA_BSA_DOM_YN AS 
SELECT * FROM MAD_BASE_STREET_ARC
WHERE DISJOINT_GEOGRAPHY NOT IN ('Y', 'N') OR DISJOINT_GEOGRAPHY IS NULL OR DISJOINT_GEOGRAPHY = ''
OR DISJOINT_RANGE NOT IN ('Y', 'N') OR DISJOINT_RANGE IS NULL OR DISJOINT_RANGE = ''
OR PAVED NOT IN ('Y', 'N') OR PAVED IS NULL OR PAVED = ''
OR PRIVATE NOT IN ('Y', 'N') OR PRIVATE IS NULL OR PRIVATE = ''
OR RAMP NOT IN ('Y', 'N') OR RAMP IS NULL OR RAMP = ''
OR BRIDGE NOT IN ('Y', 'N') OR BRIDGE IS NULL OR BRIDGE = '' 
OR TOLLWAY NOT IN ('Y', 'N') OR TOLLWAY IS NULL OR TOLLWAY = ''
OR TUNNEL NOT IN ('Y', 'N') OR TUNNEL IS NULL OR TUNNEL = ''
OR FLIPPED NOT IN ('Y', 'N') OR FLIPPED IS NULL OR FLIPPED = '';
