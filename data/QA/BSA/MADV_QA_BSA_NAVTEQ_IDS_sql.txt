# View identifies any segments in MAD_BASE_STREET_ARC that have no value in one or more
# of the following NAVTEQ-related fields... NAVTEQ_LINK_ID, NAVTEQ_FEATURE_ID, NAVTEQ_LEFT_AREA_ID,
# and NAVTEQ_RIGHT_AREA_ID.  These fields should not have NULL/blank values.
# DM - 9/26/16
# Updated 5/4/18 to remove regular expressions (irrelevant for numeric fields)

CREATE OR REPLACE VIEW MADV_QA_BSA_NAVTEQ_IDS AS 
SELECT * FROM MAD_BASE_STREET_ARC
WHERE NAVTEQ_LINK_ID IS NULL
OR NAVTEQ_FEATURE_ID IS NULL
OR NAVTEQ_LEFT_AREA_ID IS NULL 
OR NAVTEQ_RIGHT_AREA_ID IS NULL;

CREATE OR REPLACE VIEW MADV_QA_BSA_NAVTEQ_IDS AS 
SELECT * FROM MAD_BASE_STREET_ARC
WHERE (NAVTEQ_LINK_ID IS NULL OR REGEXP_LIKE (NAVTEQ_LINK_ID, '\D'))
OR NAVTEQ_FEATURE_ID IS NULL
OR (NAVTEQ_LEFT_AREA_ID IS NULL OR NAVTEQ_LEFT_AREA_ID = '' OR REGEXP_LIKE (NAVTEQ_LEFT_AREA_ID, '\D'))
OR (NAVTEQ_RIGHT_AREA_ID IS NULL OR NAVTEQ_RIGHT_AREA_ID = '' OR REGEXP_LIKE (NAVTEQ_RIGHT_AREA_ID, '\D'));
