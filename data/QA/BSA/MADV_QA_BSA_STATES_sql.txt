# View identifies any segments in MAD_BASE_STREET_ARC that have a value in STATE_LEFT_ADDRESS
# or STATE_RIGHT_ADDRESS that is outside the accepted domain.
# DM - 9/27/16

CREATE OR REPLACE VIEW MADV_QA_BSA_STATES AS 
SELECT * FROM MAD_BASE_STREET_ARC
WHERE STATE_LEFT_ADDRESS NOT IN ('RI', 'MA', 'CT', 'VT', 'NH', 'NY')
OR STATE_LEFT_ADDRESS IS NULL
OR STATE_RIGHT_ADDRESS NOT IN ('RI', 'MA', 'CT', 'VT', 'NH', 'NY')
OR STATE_RIGHT_ADDRESS IS NULL;
