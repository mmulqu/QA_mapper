# View identifies any segments in MAD_BASE_STREET_ARC that have an invalid value in LAST_EDIT_DATE.
# DM - 9/27/16
# Updated to dynamically provide current year.  11/2/17 - DM

CREATE OR REPLACE VIEW MADV_QA_BSA_EDITDATE AS
SELECT * FROM MAD_BASE_STREET_ARC BSA
WHERE TO_NUMBER(SUBSTR(LAST_EDIT_DATE, 0, 4)) NOT BETWEEN 1900 AND (SELECT TO_NUMBER(TO_CHAR(sysdate, 'YYYY')) FROM DUAL)
OR TO_NUMBER(SUBSTR(LAST_EDIT_DATE, 5, 2)) NOT BETWEEN 01 AND 12
OR TO_NUMBER(SUBSTR(LAST_EDIT_DATE, 7, 2)) NOT BETWEEN 01 AND 31
OR NOT REGEXP_LIKE(LAST_EDIT_DATE, '^[0-9]{8}$')
OR LAST_EDIT_DATE IS NULL;
