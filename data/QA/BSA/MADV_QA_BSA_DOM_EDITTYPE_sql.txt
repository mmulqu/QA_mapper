# View identifies any segments in MAD_BASE_STREET_ARC that have a value in the
# LAST_EDIT_TYPE field that is outside the accepted domain.
# DM - 10/26/17

CREATE OR REPLACE VIEW MADV_QA_BSA_DOM_EDITTYPE AS 
SELECT * FROM MAD_BASE_STREET_ARC BSA
WHERE LAST_EDIT_TYPE IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'LAST_EDIT_TYPE'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_BASE_STREET_ARC'
AND BSA.LAST_EDIT_TYPE = DOM.DOMAIN_VALUE)
ORDER BY LAST_EDIT_TYPE, COMMUNITY_ID_LEFT_ADDRESS, BASE_SEGMENT_ID;
