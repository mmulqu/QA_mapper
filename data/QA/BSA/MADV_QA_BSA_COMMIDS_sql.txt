# View identifies any segments in MAD_BASE_STREET_ARC that have a value in COMMUNITY_ID_LEFT_ADDRESS
# or COMMUNITY_ID_RIGHT_ADDRESS that is outside the accepted domain.
# DM - 9/26/16
# Updated 4/10/17 to include last OR statement, as suggested by Jack. - DM

CREATE OR REPLACE VIEW MADV_QA_BSA_COMMIDS AS 
SELECT * FROM MAD_BASE_STREET_ARC
WHERE COMMUNITY_ID_LEFT_ADDRESS NOT IN
        (SELECT DISTINCT COMMUNITY_ID FROM MAD_MSAG_COMMUNITY_POLYM)
OR COMMUNITY_ID_LEFT_ADDRESS IS NULL
OR COMMUNITY_ID_RIGHT_ADDRESS NOT IN
	(SELECT DISTINCT COMMUNITY_ID FROM MAD_MSAG_COMMUNITY_POLYM)
OR COMMUNITY_ID_RIGHT_ADDRESS IS NULL
OR (COMMUNITY_ID_LEFT_ADDRESS <> COMMUNITY_ID_RIGHT_ADDRESS
    AND BASE_SEGMENT_ID NOT IN
      (SELECT DISTINCT BRV.BASE_SEGMENT_ID
       FROM MAD_BASE_RANGE_VARIANTS BRV
       WHERE BRV.community_difference = 'S'));
