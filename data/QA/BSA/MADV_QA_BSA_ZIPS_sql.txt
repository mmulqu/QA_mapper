# View identifies any segments in MAD_BASE_STREET_ARC that have a value in ZIP_LEFT_ADDRESS
# or ZIP_RIGHT_ADDRESS that is outside the accepted domain.
# DM - 9/27/16

CREATE OR REPLACE VIEW MADV_QA_BSA_ZIPS AS 
SELECT * FROM MAD_BASE_STREET_ARC
WHERE NOT REGEXP_LIKE(ZIP_LEFT_ADDRESS, '^[0-9]{5}$')
OR ZIP_LEFT_ADDRESS IS NULL
OR NOT REGEXP_LIKE(ZIP_RIGHT_ADDRESS, '^[0-9]{5}$')
OR ZIP_RIGHT_ADDRESS IS NULL;
