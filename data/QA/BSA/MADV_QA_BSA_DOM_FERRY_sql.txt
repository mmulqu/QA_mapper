# View identifies any segments in MAD_BASE_STREET_ARC that have a value in the
# FERRY_TYPE field that is outside the accepted domain.
# DM - 9/27/16

CREATE OR REPLACE VIEW MADV_QA_BSA_DOM_FERRY AS 
SELECT * FROM MAD_BASE_STREET_ARC BSA
WHERE FERRY_TYPE IS NULL
OR FERRY_TYPE = ''
OR NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'FERRY_TYPE'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_BASE_STREET_ARC'
AND BSA.FERRY_TYPE = DOM.DOMAIN_VALUE)
ORDER BY FERRY_TYPE, COMMUNITY_ID_LEFT_ADDRESS, BASE_SEGMENT_ID;
