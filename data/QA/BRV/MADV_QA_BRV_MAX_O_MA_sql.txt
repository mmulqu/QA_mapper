# View identifies any records in MAD_MASTER_ADDRESS that have an ODD address number and that
# number is higher than the address range recorded for that street in MAD_BASE_RANGE_VARIANTS.
# Either the range in BRV needs to be extended, or the address record has an incorrect street number.
# DM - 9/15/16
# Added "BRV.PARITY_<SIDE> = '' " to parity query statements.  10/2/17 - DM
# Added regular expressions to ensure only numeric values evaluated with TO_NUMBER.  10/2/17 - DM
# Updated with REGEXP_SUBSTR functions to prevent TO_NUMBER errors in Oracle 19.  11/23/20 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_MAX_O_MA AS 
WITH BRV AS( SELECT STREET_NAME_ID, 
             MAX (GREATEST (TO_NUMBER(REGEXP_SUBSTR(TRIM(REPLACE(FROM_ADDRESS, ' ', 1)), '\d+')), 
                            TO_NUMBER(REGEXP_SUBSTR(TRIM(REPLACE(TO_ADDRESS, ' ', 1)), '\d+')))) AS HIGH_NUM
             FROM (SELECT SNV.STREET_NAME_ID, BRV.FROM_ADDRESS_LEFT AS FROM_ADDRESS, BRV.TO_ADDRESS_LEFT AS TO_ADDRESS
                   FROM MAD_BASE_RANGE_VARIANTS BRV, MAD_STREET_NAME_VARIANTS SNV
                   WHERE BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID
                   AND (BRV.PARITY_LEFT = 'O' OR BRV.PARITY_LEFT = '')
                   AND (NOT REGEXP_LIKE(BRV.TO_ADDRESS_LEFT, '\D') OR BRV.TO_ADDRESS_LEFT =  ' ')
                   AND (NOT REGEXP_LIKE(BRV.FROM_ADDRESS_LEFT, '\D') OR BRV.FROM_ADDRESS_LEFT =  ' ')
                   UNION
                   SELECT SNV.STREET_NAME_ID, BRV.FROM_ADDRESS_RIGHT AS FROM_ADDRESS, BRV.TO_ADDRESS_RIGHT AS TO_ADDRESS
                   FROM MAD_BASE_RANGE_VARIANTS BRV, MAD_STREET_NAME_VARIANTS SNV
                   WHERE BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID
                   AND (BRV.PARITY_RIGHT = 'O' OR BRV.PARITY_RIGHT = '')
                   AND (NOT REGEXP_LIKE(BRV.TO_ADDRESS_RIGHT, '\D') OR BRV.TO_ADDRESS_RIGHT =  ' ')
                   AND (NOT REGEXP_LIKE(BRV.FROM_ADDRESS_RIGHT, '\D') OR BRV.FROM_ADDRESS_RIGHT =  ' ')
                   )
            GROUP BY STREET_NAME_ID)
SELECT DISTINCT MA.MASTER_ADDRESS_ID, MA.FULL_NUMBER_STANDARDIZED AS FULL_NUM, MA.ADDRESS_NUMBER, MSN.STREET_NAME, MA.STREET_NAME_ID, 
MC.COMMUNITY_NAME, BRV.HIGH_NUM AS BRV_HIGH_NUM
FROM MAD_MASTER_ADDRESS MA, MAD_MASTER_STREET_NAME MSN, MAD_MSAG_COMMUNITY_POLYM MC, BRV, MAD_ADDRESS_POINTM AP
WHERE MA.STREET_NAME_ID = MSN.STREET_NAME_ID 
AND MA.COMMUNITY_ID = MC.COMMUNITY_ID 
AND MA.STREET_NAME_ID = BRV.STREET_NAME_ID 
AND MA.ADDRESS_POINT_ID = AP.ADDRESS_POINT_ID
AND (MA.ADDRESS_STATUS NOT IN ('INACTIVE', 'NOT_FOUND', 'DELETED') OR MA.ADDRESS_STATUS IS NULL)
AND AP.POINT_TYPE NOT IN ('PC', 'BMPC') -- ADDED BMPC POINT TYPES
AND MOD(MA.ADDRESS_NUMBER, 2) = 1
AND TO_NUMBER(BRV.HIGH_NUM) < TO_NUMBER(MA.ADDRESS_NUMBER)
ORDER BY MC.COMMUNITY_NAME, MSN.STREET_NAME, MA.FULL_NUMBER_STANDARDIZED;
