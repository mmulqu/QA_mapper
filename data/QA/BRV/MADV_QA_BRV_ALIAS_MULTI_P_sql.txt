# View identifies any records in MAD_BASE_RANGE_VARIANTS where a given BASE_SEGMENT_ID has  
# been incorrectly assigned more than one "P" (primary street) value in ALIAS_FLAG.
# THE ALIAS FLAG SHOULD NOT HAVE TWO 'P'S UNLESS THE ARC IS ON A TOWN BOUNDARY AND THE ROAD ROW IS SHARED.
# IN ALL OTHER CASES IT SHOULD BE ONE 'P' AND ZERO OR MORE 'A's OR ONE OR MORE 'C's
# 'C' ARE USED WHERE THE GEOGRAHIC COMM DOES NOT MATCH THE COMMUNITY ID ON THE BRV BUT THE ROAD IS CONSIDERED SHARED.
# DM - 9/12/16

CREATE OR REPLACE VIEW MADV_QA_BRV_ALIAS_MULTI_P AS
SELECT DISTINCT BRV.BASE_SEGMENT_ID, MSN.STREET_NAME, BRV.STREET_NAME_ID,  MSN.community_id, BRV.CNT
FROM (SELECT BASE_SEGMENT_ID, STREET_NAME_ID, COUNT(BASE_SEGMENT_ID) OVER (PARTITION BY BASE_SEGMENT_ID, ALIAS_FLAG, COMMUNITY_DIFFERENCE ORDER BY BASE_SEGMENT_ID, ALIAS_FLAG, COMMUNITY_DIFFERENCE) AS CNT
      FROM MAD_BASE_RANGE_VARIANTS
      WHERE ALIAS_FLAG = 'P'
      AND COMMUNITY_DIFFERENCE != 'S') BRV, MAD_MASTER_STREET_NAME MSN
WHERE BRV.BASE_SEGMENT_ID IN (
      SELECT BASE_SEGMENT_ID FROM (
        SELECT BASE_SEGMENT_ID, COUNT(BASE_SEGMENT_ID)
        FROM MAD_BASE_RANGE_VARIANTS
        GROUP BY BASE_SEGMENT_ID
        HAVING COUNT(BASE_SEGMENT_ID) > 1))
AND BRV.CNT > 1
AND MSN.STREET_NAME_ID = BRV.STREET_NAME_ID
ORDER BY MSN.STREET_NAME, BRV.STREET_NAME_ID, BRV.BASE_SEGMENT_ID;
