# View identifies any records in MAD_BASE_RANGE_VARIANTS with an incorrect ALIAS_FLAG value.  
# "Incorrect" means it is either blank, or is a non-null value not within the acceptable set
# of domain values.
# DM - 9/12/16

CREATE OR REPLACE VIEW MADV_QA_BRV_DOM_ALIAS AS
SELECT OBJECTID, BASE_SEGMENT_ID, ALIAS_FLAG
FROM MAD_BASE_RANGE_VARIANTS BRV
WHERE ALIAS_FLAG IS NULL
OR NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'ALIAS_FLAG'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_BASE_RANGE_VARIANTS'
AND BRV.ALIAS_FLAG = DOM.DOMAIN_VALUE)
ORDER BY ALIAS_FLAG, COMMUNITY_ID_LEFT, STREET_NAME_ID, BASE_SEGMENT_ID;
