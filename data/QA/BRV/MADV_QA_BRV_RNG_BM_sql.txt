# View identifies any records in MAD_BASE_RANGE_VARIANTS that have an ADDRESS_RANGE_TYPE of
# 'B' or 'M', but one or more values in the address range fields of that record are not populated.
# DM - 9/12/16

CREATE OR REPLACE VIEW MADV_QA_BRV_RNG_BM AS
SELECT OBJECTID, BASE_RANGE_VARIANT_ID, ADDRESS_RANGE_TYPE, 
FROM_ADDRESS_LEFT, TO_ADDRESS_LEFT, FROM_ADDRESS_RIGHT, TO_ADDRESS_RIGHT FROM MAD_BASE_RANGE_VARIANTS
WHERE ADDRESS_RANGE_TYPE IN ('B','M')
MINUS
SELECT OBJECTID, BASE_RANGE_VARIANT_ID, ADDRESS_RANGE_TYPE, 
FROM_ADDRESS_LEFT, TO_ADDRESS_LEFT, FROM_ADDRESS_RIGHT, TO_ADDRESS_RIGHT FROM MAD_BASE_RANGE_VARIANTS
WHERE (REGEXP_LIKE(FROM_ADDRESS_LEFT, '\d') AND REGEXP_LIKE(TO_ADDRESS_LEFT, '\d'))
OR (REGEXP_LIKE(FROM_ADDRESS_RIGHT, '\d') AND REGEXP_LIKE(TO_ADDRESS_RIGHT, '\d'));
