# View identifies any records in MAD_BASE_RANGE_VARIANTS where any of the address range fields 
# contain a value that is not an integer or a space.  Only integers and spaces are acceptable values.
# DM - 9/14/16

CREATE OR REPLACE VIEW MADV_QA_BRV_NUM_REGEXP AS
SELECT OBJECTID, BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID, FROM_ADDRESS_LEFT, TO_ADDRESS_LEFT, FROM_ADDRESS_RIGHT, TO_ADDRESS_RIGHT
FROM MAD_BASE_RANGE_VARIANTS
WHERE (REGEXP_LIKE(FROM_ADDRESS_LEFT, '\D') AND FROM_ADDRESS_LEFT <> ' ')
OR (REGEXP_LIKE(TO_ADDRESS_LEFT, '\D') AND TO_ADDRESS_LEFT <> ' ')
OR (REGEXP_LIKE(FROM_ADDRESS_RIGHT, '\D') AND FROM_ADDRESS_RIGHT <> ' ')
OR (REGEXP_LIKE(TO_ADDRESS_RIGHT, '\D') AND TO_ADDRESS_RIGHT <> ' ');
