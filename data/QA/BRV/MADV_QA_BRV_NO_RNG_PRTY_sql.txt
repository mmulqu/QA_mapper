# View identifies any records in MAD_BASE_RANGE_VARIANTS that have null address range values
# but a non-null address range parity value.  Parity can't exist without associated address ranges.
# DM - 9/13/16

CREATE OR REPLACE VIEW MADV_QA_BRV_NO_RNG_PRTY AS
SELECT OBJECTID, BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID, ADDRESS_RANGE_TYPE, FROM_ADDRESS_LEFT, TO_ADDRESS_LEFT,
PARITY_LEFT, FROM_ADDRESS_RIGHT, TO_ADDRESS_RIGHT, PARITY_RIGHT
FROM MAD_BASE_RANGE_VARIANTS
WHERE ((FROM_ADDRESS_LEFT = ' ' OR TO_ADDRESS_LEFT = ' ') AND PARITY_LEFT <> ' ')
OR ((FROM_ADDRESS_RIGHT = ' ' OR TO_ADDRESS_RIGHT = ' ') AND PARITY_RIGHT <> ' ');
