# View identifies any records in MAD_BASE_RANGE_VARIANTS with an unacceptable COMMUNITY_DIFFERENCE value.
# "Unacceptable" means it falls outside the accepted domain of non-null values, or is blank/NULL.
# DM - 9/14/16
# Updated to use domain values in MADMETA_DOMAINS table instead.  3/17/22 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_COMMDIFF AS
SELECT OBJECTID, BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID, COMMUNITY_DIFFERENCE
FROM MAD_BASE_RANGE_VARIANTS
WHERE COMMUNITY_DIFFERENCE NOT IN 
  (SELECT DISTINCT DOMAIN_VALUE 
   FROM MAD.MADMETA_DOMAINS 
   WHERE ENTITY_WITH_DOMAIN = 'MAD_BASE_RANGE_VARIANTS' 
     AND DOMAIN_FIELD = 'COMMUNITY_DIFFERENCE'
  ) -- NONE, SHARED, DIFFERENT, OVERLAP (DUPLICATE ADDRESSES), INTERSECTION 
OR COMMUNITY_DIFFERENCE IS NULL;
