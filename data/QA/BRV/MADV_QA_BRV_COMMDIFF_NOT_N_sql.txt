# View identifies any records in MAD_BASE_RANGE_VARIANTS with a COMMUNITY_DIFFERENCE value that shouldn't be 'N', but is.
# DM - 9/15/17
# Updated to include "UNASSIGNED" addresses as excluded from comparison.  12/12/17 - DM
# Updated 1/24/18 to compare street name IDs by using street name variants instead. (SQL provided by Jack) DM
# (previously used WHERE BRV.STREET_NAME_ID = MA.STREET_NAME_ID)
# Updated 2/16/18.  Added (OR SNV.STREET_NAME_ID != BRV.STREET_NAME_ID)
# Updated again from Jack's email 12/20/18.  -DM
# Updated with REGEXP_SUBSTR functions to prevent TO_NUMBER errors in Oracle 19.  11/23/20 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_COMMDIFF_NOT_N AS
SELECT DISTINCT BRV.BASE_RANGE_VARIANT_ID, BRV.COMMUNITY_DIFFERENCE, SNID_MSN.STREET_NAME_ID, SNID_MSN.STREET_NAME, SNID_MSN.COMMUNITY_ID, SNVID_MSN.STREET_NAME_ID AS SNV_STREET_NAME_ID, SNVID_MSN.STREET_NAME AS SNV_STREET_NAME, 
  BRV.COMMUNITY_ID_LEFT AS BRV_COMMUNITY_ID, BRV.FROM_ADDRESS_LEFT, BRV.TO_ADDRESS_LEFT, BRV.FROM_ADDRESS_RIGHT, BRV.TO_ADDRESS_RIGHT, MA.COMMUNITY_ID AS MA_COMMUNITY_ID
  FROM MAD_BASE_RANGE_VARIANTS BRV 
LEFT JOIN MAD_STREET_NAME_VARIANTS SNV ON BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID
LEFT JOIN MAD_MASTER_STREET_NAME SNVID_MSN ON SNVID_MSN.STREET_NAME_ID = SNV.STREET_NAME_ID
LEFT JOIN MAD_MASTER_ADDRESS MA ON SNVID_MSN.STREET_NAME_ID = MA.STREET_NAME_ID
LEFT JOIN MAD_MASTER_STREET_NAME SNID_MSN ON SNID_MSN.STREET_NAME_ID = BRV.STREET_NAME_ID
WHERE BRV.COMMUNITY_DIFFERENCE NOT IN ('D', 'S', 'O')
AND 
  (
    (MA.ADDRESS_STATUS NOT IN ('INACTIVE', 'NOT_FOUND', 'UNASSIGNED') OR MA.ADDRESS_STATUS IS NULL)
    AND MA.GEOGRAPHIC_TOWN_ID != MA.ADDRESS_TOWN_ID 
    AND
      (
            (BRV.PARITY_LEFT = 'E'
             AND MOD(TO_NUMBER(MA.ADDRESS_NUMBER ), 2) = 0
             AND BRV.FROM_ADDRESS_LEFT <> ' '
             AND BRV.TO_ADDRESS_LEFT <> ' '
             AND ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(REGEXP_SUBSTR(BRV.FROM_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(BRV.TO_ADDRESS_LEFT, '\d+')))
                   OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(REGEXP_SUBSTR(BRV.TO_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(BRV.FROM_ADDRESS_LEFT, '\d+'))))
             )
           OR  
            (BRV.PARITY_LEFT = 'O'
             AND MOD(TO_NUMBER(MA.ADDRESS_NUMBER ), 2) = 1
             AND BRV.FROM_ADDRESS_LEFT <> ' '
             AND BRV.TO_ADDRESS_LEFT <> ' '
             AND ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(REGEXP_SUBSTR(BRV.FROM_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(BRV.TO_ADDRESS_LEFT, '\d+')))
                   OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(REGEXP_SUBSTR(BRV.TO_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(BRV.FROM_ADDRESS_LEFT, '\d+'))))
            )
           OR  
            (BRV.PARITY_RIGHT = 'E'
             AND MOD(TO_NUMBER(MA.ADDRESS_NUMBER ), 2) = 0
             AND BRV.FROM_ADDRESS_RIGHT <> ' '
             AND BRV.TO_ADDRESS_RIGHT <> ' '
             AND ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(REGEXP_SUBSTR(BRV.FROM_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(BRV.TO_ADDRESS_RIGHT, '\d+')))
                   OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(REGEXP_SUBSTR(BRV.TO_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(BRV.FROM_ADDRESS_RIGHT, '\d+'))))
            )
           OR  
            (BRV.PARITY_RIGHT = 'O'
             AND MOD(TO_NUMBER(MA.ADDRESS_NUMBER ), 2) = 1
             AND BRV.FROM_ADDRESS_RIGHT <> ' '
             AND BRV.TO_ADDRESS_RIGHT <> ' '
             AND ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(REGEXP_SUBSTR(BRV.FROM_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(BRV.TO_ADDRESS_RIGHT, '\d+')))
                   OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(REGEXP_SUBSTR(BRV.TO_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(BRV.FROM_ADDRESS_RIGHT, '\d+'))))
        )
      )
    OR (BRV.COMMUNITY_ID_LEFT != SNID_MSN.COMMUNITY_ID)
  )
ORDER BY BRV.COMMUNITY_ID_LEFT, SNVID_MSN.STREET_NAME, BRV.TO_ADDRESS_LEFT, BRV.TO_ADDRESS_RIGHT;

CREATE OR REPLACE VIEW MADV_QA_BRV_COMMDIFF_NOT_N AS
SELECT BRV.BASE_RANGE_VARIANT_ID, brv.from_address_left, brv.to_address_left, brv.from_address_right, brv.to_address_right, 
       brv.community_difference, MA.full_number_standardized, ma.geographic_town_id, ma.address_town_id, ma.master_address_id
FROM MAD_BASE_RANGE_VARIANTS BRV, MAD_MASTER_ADDRESS MA, MAD_STREET_NAME_VARIANTS SNV, MAD_MASTER_STREET_NAME MSN
WHERE BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID
AND SNV.STREET_NAME_ID = MSN.STREET_NAME_ID
AND MSN.STREET_NAME_ID = MA.STREET_NAME_ID
AND MA.ADDRESS_STATUS NOT IN ('INACTIVE', 'NOT_FOUND', 'UNASSIGNED')
AND (MA.GEOGRAPHIC_TOWN_ID != MA.ADDRESS_TOWN_ID OR SNV.STREET_NAME_ID != BRV.STREET_NAME_ID)
AND BRV.COMMUNITY_DIFFERENCE NOT IN ('D', 'S', 'O')
AND(
    (BRV.PARITY_LEFT = 'E'
     AND MOD(TO_NUMBER(MA.ADDRESS_NUMBER ), 2) = 0
     AND BRV.FROM_ADDRESS_LEFT <> ' '
     AND BRV.TO_ADDRESS_LEFT <> ' '
     AND ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(TRIM(BRV.FROM_ADDRESS_LEFT)) AND TO_NUMBER(TRIM(BRV.TO_ADDRESS_LEFT)))
           OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(TRIM(BRV.TO_ADDRESS_LEFT)) AND TO_NUMBER(TRIM(BRV.FROM_ADDRESS_LEFT))))
     )
   OR  
    (BRV.PARITY_LEFT = 'O'
     AND MOD(TO_NUMBER(MA.ADDRESS_NUMBER ), 2) = 1
     AND BRV.FROM_ADDRESS_LEFT <> ' '
     AND BRV.TO_ADDRESS_LEFT <> ' '
     AND ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(TRIM(BRV.FROM_ADDRESS_LEFT)) AND TO_NUMBER(TRIM(BRV.TO_ADDRESS_LEFT)))
           OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(TRIM(BRV.TO_ADDRESS_LEFT)) AND TO_NUMBER(TRIM(BRV.FROM_ADDRESS_LEFT))))
    )
   OR  
    (BRV.PARITY_RIGHT = 'E'
     AND MOD(TO_NUMBER(MA.ADDRESS_NUMBER ), 2) = 0
     AND BRV.FROM_ADDRESS_RIGHT <> ' '
     AND BRV.TO_ADDRESS_RIGHT <> ' '
     AND ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(TRIM(BRV.FROM_ADDRESS_RIGHT)) AND TO_NUMBER(TRIM(BRV.TO_ADDRESS_RIGHT)))
           OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(TRIM(BRV.TO_ADDRESS_RIGHT)) AND TO_NUMBER(TRIM(BRV.FROM_ADDRESS_RIGHT))))
    )
   OR  
    (BRV.PARITY_RIGHT = 'O'
     AND MOD(TO_NUMBER(MA.ADDRESS_NUMBER ), 2) = 1
     AND BRV.FROM_ADDRESS_RIGHT <> ' '
     AND BRV.TO_ADDRESS_RIGHT <> ' '
     AND ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(TRIM(BRV.FROM_ADDRESS_RIGHT)) AND TO_NUMBER(TRIM(BRV.TO_ADDRESS_RIGHT)))
           OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(TRIM(BRV.TO_ADDRESS_RIGHT)) AND TO_NUMBER(TRIM(BRV.FROM_ADDRESS_RIGHT))))
    )
)
ORDER BY MA.GEOGRAPHIC_TOWN_ID, MA.FULL_NUMBER_STANDARDIZED;
