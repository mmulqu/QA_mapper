# View identifies any records in MAD_BASE_RANGE_VARIANTS with a mixed parity address range
# that has been incorrectly assigned "P" instead of "A" in ALIAS_FLAG.
# EVERY M RANGE SHOULD BE AN A
# THERE ARE A HANDFUL OF RECORDS WHERE THE 'M' RECORD IS A 'P' AND SHOULD BE BECAUSE THERE IS NO OTHER RANGE FOR THAT PARTICULAR ARC.
# THESE RECORDS SHOULD HAVE A BRV ADDED WITH NO RANGE.
# DM - 9/12/16

CREATE OR REPLACE VIEW MADV_QA_BRV_ALIAS_PARITYMIX AS
SELECT OBJECTID, BASE_SEGMENT_ID, ALIAS_FLAG, ADDRESS_RANGE_TYPE FROM MAD_BASE_RANGE_VARIANTS
WHERE ALIAS_FLAG = 'P'
AND ADDRESS_RANGE_TYPE = 'M'
AND BASE_SEGMENT_ID IN 
  (SELECT DISTINCT BRV.BASE_SEGMENT_ID FROM 
    (SELECT BASE_SEGMENT_ID, STREET_NAME_ID, COUNT(BASE_SEGMENT_ID) OVER (PARTITION BY BASE_SEGMENT_ID, ALIAS_FLAG ORDER BY BASE_SEGMENT_ID, ALIAS_FLAG) AS CNT
     FROM MAD_BASE_RANGE_VARIANTS
     WHERE ALIAS_FLAG = 'P') BRV, MAD_MASTER_STREET_NAME MSN
   WHERE BRV.BASE_SEGMENT_ID IN
      (SELECT BASE_SEGMENT_ID FROM
         (SELECT BASE_SEGMENT_ID, COUNT(BASE_SEGMENT_ID)
          FROM MAD_BASE_RANGE_VARIANTS
          GROUP BY BASE_SEGMENT_ID
          HAVING COUNT(BASE_SEGMENT_ID) > 1))
   AND BRV.CNT > 1
   AND MSN.STREET_NAME_ID = BRV.STREET_NAME_ID)
ORDER BY STREET_NAME_ID, BASE_SEGMENT_ID;
