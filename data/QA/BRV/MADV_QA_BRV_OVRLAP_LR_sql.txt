# View identifies any records in MAD_BASE_RANGE_VARIANTS with overlapping address ranges on only 
# OPPOSITE sides of all streets.
# DM - 1/30/17
# Added "AND (S1.STREET_NAME_ID NOT IN (SELECT STREET_NAME_ID FROM MAD_MASTER_STREET_NAME WHERE STREET_NAME IS NULL))" - 4/20/17 DM
# Re-incorporated Boston data - 5/11/17
# Added extra REGEXP to cover all four street number fields to avoid "Invalid Number" error. - 10/4/17 - DM
# Updated with REGEXP_SUBSTR functions to prevent TO_NUMBER errors in Oracle 19.  11/23/20 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_OVRLAP_LR AS            
WITH S1 AS -- WITH CLAUSES ADDED FOR CLARITY AND SPEED.
    (SELECT MSN.STREET_NAME_ID, MSN.street_name, MSN.community_id, BRV.FROM_ADDRESS_LEFT, BRV.TO_ADDRESS_LEFT, BRV.FROM_ADDRESS_RIGHT, 
            BRV.TO_ADDRESS_RIGHT, BRV.COMMUNITY_ID_LEFT, BRV.COMMUNITY_ID_RIGHT,BRV.BASE_SEGMENT_ID, BRV.community_difference,
            BRV.street_name_variant_id, BRV.PARITY_LEFT, BRV.PARITY_RIGHT
     FROM MAD_BASE_RANGE_VARIANTS BRV
     JOIN MAD_STREET_NAME_VARIANTS SNV ON BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID -- SNID IS OBTAINED THROUGH THE SNVID SO THAT 'FALSE' OVERLAPS ARE NOT IDENTIFIED WHEN THE SNVID POINTS TO A DIFFERENT SNID. FOR EXAMPLE THE ADJACENT TOWN USES A DIFFERENT STREET NAME.
     JOIN MAD_MASTER_STREET_NAME MSN ON SNV.STREET_NAME_ID = MSN.STREET_NAME_ID),
S2 AS
    (SELECT MSN.STREET_NAME_ID, MSN.street_name, MSN.community_id, BRV.FROM_ADDRESS_LEFT, BRV.TO_ADDRESS_LEFT, BRV.FROM_ADDRESS_RIGHT, 
            BRV.TO_ADDRESS_RIGHT, BRV.COMMUNITY_ID_LEFT, BRV.COMMUNITY_ID_RIGHT,BRV.BASE_SEGMENT_ID, BRV.community_difference,
            BRV.street_name_variant_id, BRV.PARITY_LEFT, BRV.PARITY_RIGHT
     FROM MAD_BASE_RANGE_VARIANTS BRV
     JOIN MAD_STREET_NAME_VARIANTS SNV ON BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID
     JOIN MAD_MASTER_STREET_NAME MSN ON SNV.STREET_NAME_ID = MSN.STREET_NAME_ID)
SELECT DISTINCT S1.BASE_SEGMENT_ID AS BASE_SEGMENT1, S1.STREET_NAME STREET_NAME1, S1.FROM_ADDRESS_LEFT FROM_ADDNUM_L1, S1.TO_ADDRESS_LEFT TO_ADDNUM_L1, S1.PARITY_LEFT PARITY_L1,
S1.FROM_ADDRESS_RIGHT FROM_ADDNUM_R1, S1.TO_ADDRESS_RIGHT TO_ADDNUM_R1, S1.PARITY_RIGHT PARITY_R1, S1.COMMUNITY_ID_LEFT COMM_ID_LEFT1, S1.COMMUNITY_ID_RIGHT COMM_ID_RIGHT1, S1.COMMUNITY_DIFFERENCE COMM_DIFF1, 
S2.BASE_SEGMENT_ID AS OVERLAPS_WITH, S2.STREET_NAME STREET_NAME2, S2.FROM_ADDRESS_LEFT FROM_ADDNUM_L2, S2.TO_ADDRESS_LEFT TO_ADDNUM_L2, S2.PARITY_LEFT PARITY_L2,
S2.FROM_ADDRESS_RIGHT FROM_ADDNUM_R2, S2.TO_ADDRESS_RIGHT TO_ADDNUM_R2, S2.PARITY_RIGHT PARITY_R2, S2.COMMUNITY_ID_LEFT COMM_ID_LEFT2, S2.COMMUNITY_ID_RIGHT COMM_ID_RIGHT2, S2.COMMUNITY_DIFFERENCE COMM_DIFF2
FROM S1
INNER JOIN S2
ON S1.STREET_NAME_ID = S2.STREET_NAME_ID -- STEP THROUGH SNVID
WHERE (
        (TO_NUMBER(REGEXP_SUBSTR(S1.FROM_ADDRESS_LEFT, '\d+'))  BETWEEN TO_NUMBER(REGEXP_SUBSTR(S2.TO_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(S2.FROM_ADDRESS_RIGHT, '\d+')) 
         OR TO_NUMBER(REGEXP_SUBSTR(S1.TO_ADDRESS_LEFT, '\d+'))  BETWEEN TO_NUMBER(REGEXP_SUBSTR(S2.TO_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(S2.FROM_ADDRESS_RIGHT, '\d+')) 
         OR TO_NUMBER(REGEXP_SUBSTR(S2.FROM_ADDRESS_RIGHT, '\d+'))  BETWEEN TO_NUMBER(REGEXP_SUBSTR(S1.TO_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(S1.FROM_ADDRESS_LEFT, '\d+')) 
         OR TO_NUMBER(REGEXP_SUBSTR(S2.TO_ADDRESS_RIGHT, '\d+'))  BETWEEN TO_NUMBER(REGEXP_SUBSTR(S1.TO_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(S1.FROM_ADDRESS_LEFT, '\d+')))
      OR
        (TO_NUMBER(REGEXP_SUBSTR(S1.FROM_ADDRESS_LEFT, '\d+'))  BETWEEN TO_NUMBER(REGEXP_SUBSTR(S2.FROM_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(S2.TO_ADDRESS_RIGHT, '\d+')) 
         OR TO_NUMBER(REGEXP_SUBSTR(S1.TO_ADDRESS_LEFT, '\d+'))  BETWEEN TO_NUMBER(REGEXP_SUBSTR(S2.FROM_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(S2.TO_ADDRESS_RIGHT, '\d+')) 
         OR TO_NUMBER(REGEXP_SUBSTR(S2.FROM_ADDRESS_RIGHT, '\d+'))  BETWEEN TO_NUMBER(REGEXP_SUBSTR(S1.FROM_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(S1.TO_ADDRESS_LEFT, '\d+')) 
         OR TO_NUMBER(REGEXP_SUBSTR(S2.TO_ADDRESS_RIGHT, '\d+'))  BETWEEN TO_NUMBER(REGEXP_SUBSTR(S1.FROM_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(S1.TO_ADDRESS_LEFT, '\d+')))
       )
AND (MOD(TO_NUMBER(REGEXP_SUBSTR(S1.FROM_ADDRESS_LEFT, '\d+')), 2) = MOD(TO_NUMBER(REGEXP_SUBSTR(S2.FROM_ADDRESS_RIGHT, '\d+')), 2) 
     OR MOD(TO_NUMBER(REGEXP_SUBSTR(S1.TO_ADDRESS_LEFT, '\d+')), 2) = MOD(TO_NUMBER(REGEXP_SUBSTR(S2.TO_ADDRESS_RIGHT, '\d+')), 2))
AND REGEXP_LIKE(S1.FROM_ADDRESS_LEFT, '^\d+$')
AND REGEXP_LIKE(S1.TO_ADDRESS_LEFT, '^\d+$')
AND REGEXP_LIKE(S2.FROM_ADDRESS_RIGHT, '^\d+$')
AND REGEXP_LIKE(S2.TO_ADDRESS_RIGHT, '^\d+$')
AND S1.BASE_SEGMENT_ID <> S2.BASE_SEGMENT_ID
-- TOGGLED TO INCLUDE BOSTON COMMUNITIES AS OF 20170511:
-- AND S1.COMMUNITY_ID_LEFT NOT IN (37, 45, 58, 85, 92, 155, 158, 189, 271, 274, 296, 352, 381) -- BECAUSE WE HAVE 'CLEANED UP' THE L/R COMMS IN BRV ONLY THE LEFT SIDE HAS TO BE CHECKED.
AND (S1.COMMUNITY_DIFFERENCE != 'O' OR S1.COMMUNITY_DIFFERENCE IS NULL)
AND (S2.COMMUNITY_DIFFERENCE != 'O' OR S1.COMMUNITY_DIFFERENCE IS NULL)
AND (S1.STREET_NAME_ID NOT IN (SELECT STREET_NAME_ID FROM MAD_MASTER_STREET_NAME WHERE STREET_NAME IS NULL))
-- AND (S1.PARITY_LEFT = 'M' OR S2.PARITY_RIGHT = 'M')
ORDER BY S1.STREET_NAME, S1.BASE_SEGMENT_ID;
