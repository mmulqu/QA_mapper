# View identifies any records in MAD_BASE_RANGE_VARIANTS with an incorrect ROUTE_TYPE value.  
# "Incorrect" means it contains a non-null value not within the acceptable set of domain values.
# DM - 10/25/17

CREATE OR REPLACE VIEW MADV_QA_BRV_DOM_RT_TYPE AS
SELECT OBJECTID, BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID, ROUTE_TYPE
FROM MAD_BASE_RANGE_VARIANTS BRV
WHERE ROUTE_TYPE IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'ROUTE_TYPE'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_BASE_RANGE_VARIANTS'
AND BRV.ROUTE_TYPE = DOM.DOMAIN_VALUE)
ORDER BY COMMUNITY_ID_LEFT, STREET_NAME_ID, BASE_SEGMENT_ID, ROUTE_TYPE;
