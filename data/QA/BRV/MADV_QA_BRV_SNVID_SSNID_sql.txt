# View identifies any records in MAD_BASE_RANGE_VARIANTS that have a STREETNAME_SOURCE_NAME_ID not in sync
# with the SOURCE_NAME_ID of associated records in MAD_STREET_NAME_VARIANTS.
# DM - 9/12/16
# 2/3/17 - This QA check was inactivated.

CREATE OR REPLACE VIEW MADV_QA_BRV_SNVID_SSNID AS
SELECT BRV.OBJECTID, BRV.BASE_SEGMENT_ID, BRV.STREET_NAME_VARIANT_ID, BRV.STREETNAME_SOURCE_NAME_ID, SNV.SOURCE_NAME_ID AS SNV_SOURCE_NAME_ID
FROM MAD_BASE_RANGE_VARIANTS BRV, MAD_STREET_NAME_VARIANTS SNV
WHERE SNV.STREET_NAME_VARIANT_ID = BRV.STREET_NAME_VARIANT_ID
AND BRV.STREETNAME_SOURCE_NAME_ID <> (
    SELECT SOURCE_NAME_ID
    FROM MAD_STREET_NAME_VARIANTS SNV
    WHERE SNV.STREET_NAME_VARIANT_ID = BRV.STREET_NAME_VARIANT_ID)
ORDER BY BRV.STREET_NAME_VARIANT_ID, BRV.BASE_SEGMENT_ID, BRV.OBJECTID;
