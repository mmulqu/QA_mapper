# View identifies any records in MAD_BASE_RANGE_VARIANTS where primary streets and their associated 
# MAD_STREET_NAME_VARIANTS records have a different STREET_NAME_ID than the one in base range variants.
# DM - 9/12/16
# Added "AND SNV.SNID_FLAG IS NULL" to incorporate recent changes to the master street name table being cleaned up.  DM - 9/27/21

CREATE OR REPLACE VIEW MADV_QA_BRV_SNVID_SNID AS
SELECT BRV.OBJECTID, BRV.BASE_SEGMENT_ID, MSN1.STREET_NAME_ID AS BRV_STREET_NAME_ID, MSN2.STREET_NAME_ID AS SNV_STREET_NAME_ID,  
MSN1.STREET_NAME AS BRV_STREET_NAME, MSN2.STREET_NAME AS SNV_STREET_NAME, MSN1.COMMUNITY_ID as BRV_COMMUNITY_ID, MSN2.COMMUNITY_ID AS SNV_COMMUNITY_ID
FROM MAD_BASE_RANGE_VARIANTS BRV, MAD_STREET_NAME_VARIANTS SNV, MAD_MASTER_STREET_NAME MSN1, MAD_MASTER_STREET_NAME MSN2
WHERE BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID
AND BRV.STREET_NAME_ID != SNV.STREET_NAME_ID
AND BRV.STREET_NAME_ID = MSN1.STREET_NAME_ID
AND SNV.STREET_NAME_ID = MSN2.STREET_NAME_ID
AND BRV.ALIAS_FLAG <> 'A'
AND SNV.SNID_FLAG IS NULL 
ORDER BY MSN1.COMMUNITY_ID, MSN1.STREET_NAME_ID, BRV.BASE_SEGMENT_ID;
