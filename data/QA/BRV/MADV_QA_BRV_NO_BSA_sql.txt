# View identifies any records in MAD_BASE_RANGE_VARIANTS with an incorrect BASE_SEGMENT_ID.  "Incorrect" means it is either
# blank, or has an ID that doesn't exist in MAD_BASE_SEGMENT_ARC.
# DM - 9/7/16

CREATE OR REPLACE VIEW MADV_QA_BRV_NO_BSA AS
SELECT * FROM MAD_BASE_RANGE_VARIANTS BRV
WHERE BASE_SEGMENT_ID IS NULL
OR REGEXP_LIKE(BASE_SEGMENT_ID, '^ +$')
OR BASE_SEGMENT_ID = ' '
OR BASE_SEGMENT_ID = ''
OR NOT EXISTS (SELECT * FROM MAD_BASE_STREET_ARC BA
WHERE BA.BASE_SEGMENT_ID = BRV.BASE_SEGMENT_ID)
ORDER BY COMMUNITY_ID_LEFT, COMMUNITY_ID_RIGHT, STREET_NAME_ID;
