# View identifies any records in MAD_BASE_RANGE_VARIANTS with an incorrect STREET_NAME_VARIANT_ID.  
# "Incorrect" means it is either blank, or has an ID that doesn't exist in MAD_STREET_NAME_VARIANTS.
# DM - 9/12/16

CREATE OR REPLACE VIEW MADV_QA_BRV_SNVID_XIST AS
SELECT OBJECTID, BASE_SEGMENT_ID, STREET_NAME_VARIANT_ID FROM MAD_BASE_RANGE_VARIANTS BRV
WHERE NOT EXISTS (SELECT * FROM MAD_STREET_NAME_VARIANTS SNV
  WHERE SNV.STREET_NAME_VARIANT_ID = BRV.STREET_NAME_VARIANT_ID)
OR STREET_NAME_VARIANT_ID IS NULL
OR STREET_NAME_VARIANT_ID  = '';
