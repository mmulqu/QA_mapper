# View identifies any records in MAD_BASE_RANGE_VARIANTS referencing a street with geometry that 
# incorrectly has a REAL_STREET value of 'A', 'E', 'P', or 'M' in MAD_MASTER_STREET_NAME.
# DM - 9/13/16

CREATE OR REPLACE VIEW MADV_QA_BRV_REALSTREET AS
SELECT BRV.OBJECTID, BRV.BASE_SEGMENT_ID, BRV.BASE_RANGE_VARIANT_ID, MSN.STREET_NAME_ID, MSN.STREET_NAME, MSN.COMMUNITY_ID, MSN.REAL_STREET
FROM MAD_BASE_RANGE_VARIANTS BRV, MAD_MASTER_STREET_NAME MSN
WHERE BRV.STREET_NAME_ID = MSN.STREET_NAME_ID
AND MSN.REAL_STREET IN ('A', 'E', 'P', 'M')
ORDER BY MSN.COMMUNITY_ID, MSN.STREET_NAME;
