# View identifies any records in MAD_BASE_RANGE_VARIANTS with an unpopulated LINK_FEAT_ID.
# DM - 9/12/16

CREATE OR REPLACE VIEW MADV_QA_BRV_LINKFEAT AS
SELECT OBJECTID, BASE_SEGMENT_ID, LINK_FEAT_ID
FROM MAD_BASE_RANGE_VARIANTS
WHERE LINK_FEAT_ID IS NULL
OR LINK_FEAT_ID = ' '
OR LINK_FEAT_ID = ''
ORDER BY COMMUNITY_ID_LEFT, STREET_NAME_ID, BASE_SEGMENT_ID;
