# View identifies any records in MAD_BASE_RANGE_VARIANTS where overlapping address ranges are found 
# on the same arc on the opposite site of the street. (in different BRV records)
# Excludes BOSTON.
# DM - 9/15/16
# Updated 2/1/17 so both ascending and descending ranges are included. - DM
# Re-incorporated Boston communities by commenting out exclusion statement.  10/19/17 - DM
# Updated with REGEXP_SUBSTR functions to prevent TO_NUMBER errors in Oracle 19.  11/23/20 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_OVRLAP_ARC AS 
SELECT DISTINCT OBJECTID, BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID, STREET_NAME_ID, COMMUNITY_ID_LEFT, FROM_ADDRESS_LEFT, TO_ADDRESS_LEFT,
FROM_ADDRESS_RIGHT, TO_ADDRESS_RIGHT
FROM MAD_BASE_RANGE_VARIANTS
WHERE (
      (TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')) BETWEEN TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+'))
       OR TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')) BETWEEN TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+'))
       OR TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+')) BETWEEN TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+'))
       OR TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+')) BETWEEN TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')))
      OR
      (TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')) BETWEEN TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+'))
      OR TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')) BETWEEN TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+'))
      OR TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+')) BETWEEN TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+'))
      OR TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+')) BETWEEN TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')) AND TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')))
      )
AND REGEXP_LIKE(FROM_ADDRESS_RIGHT, '^\d+$') AND REGEXP_LIKE(FROM_ADDRESS_LEFT, '^\d+$')
AND REGEXP_LIKE(TO_ADDRESS_RIGHT, '^\d+$') AND REGEXP_LIKE(TO_ADDRESS_LEFT, '^\d+$')
AND (MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')), 2) = MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+')), 2) 
     OR MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')), 2) = MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+')), 2))
--AND COMMUNITY_ID_LEFT NOT IN (37, 45, 58, 85, 92, 155, 158, 189, 271, 274, 296, 352, 381)
ORDER BY COMMUNITY_ID_LEFT, street_name_id, base_segment_id;
