# View identifies any records in MAD_BASE_RANGE_VARIANTS with an unacceptable PARITY_LEFT or PARITY_RIGHT
# value. "Unacceptable" means it falls outside the accepted domain of non-null values, or is blank/NULL.
# DM - 9/14/16

CREATE OR REPLACE VIEW MADV_QA_BRV_DOM_PARITY AS            
SELECT OBJECTID, BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID, PARITY_LEFT, PARITY_RIGHT FROM MAD_BASE_RANGE_VARIANTS BRV
WHERE PARITY_LEFT IS NULL
OR PARITY_RIGHT IS NULL
OR (NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
                WHERE DOM.DOMAIN_FIELD = 'PARITY_LEFT'
                AND DOM.ENTITY_WITH_DOMAIN = 'MAD_BASE_RANGE_VARIANTS'
                AND BRV.PARITY_LEFT = DOM.DOMAIN_VALUE)
    AND PARITY_LEFT <> ' ')
OR (NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
                WHERE DOM.DOMAIN_FIELD = 'PARITY_RIGHT'
                AND DOM.ENTITY_WITH_DOMAIN = 'MAD_BASE_RANGE_VARIANTS'
                AND BRV.PARITY_RIGHT = DOM.DOMAIN_VALUE)
    AND PARITY_RIGHT <> ' ')
ORDER BY BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID;
