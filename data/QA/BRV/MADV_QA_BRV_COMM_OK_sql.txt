# View identifies any records in MAD_BASE_RANGE_VARIANTS with an unacceptable community ID.
# "Unacceptable" means it falls outside the accepted domain of non-null values, or is blank/NULL.
# DM - 9/14/16
# Updated to include BOSTON comms as an exclusion.  10/20/16 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_COMM_OK AS            
SELECT OBJECTID, BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID, COMMUNITY_ID_LEFT, COMMUNITY_ID_RIGHT FROM MAD_BASE_RANGE_VARIANTS
WHERE ((COMMUNITY_ID_LEFT IS NOT NULL AND COMMUNITY_ID_LEFT NOT IN 
          (SELECT DISTINCT COMMUNITY_ID FROM MAD_MSAG_COMMUNITY_POLYM)
           AND COMMUNITY_ID_LEFT != ALL (37, 45, 58, 85, 92, 155, 158, 189, 271, 274, 296, 352, 381))
      OR COMMUNITY_ID_LEFT IS NULL
      OR COMMUNITY_ID_LEFT = '')
OR ((COMMUNITY_ID_RIGHT IS NOT NULL AND COMMUNITY_ID_RIGHT NOT IN 
          (SELECT DISTINCT COMMUNITY_ID FROM MAD_MSAG_COMMUNITY_POLYM)
           AND COMMUNITY_ID_RIGHT != ALL (37, 45, 58, 85, 92, 155, 158, 189, 271, 274, 296, 352, 381))
      OR COMMUNITY_ID_RIGHT IS NULL
      OR COMMUNITY_ID_RIGHT = '');
