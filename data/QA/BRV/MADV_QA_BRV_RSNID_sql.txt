# View identifies any records in MAD_BASE_RANGE_VARIANTS with an incorrect RANGE_SOURCE_NAME_ID.  
# "Incorrect" means it is either blank, or has an ID that doesn't exist in MAD_SOURCE.
# DM - 9/7/16

CREATE OR REPLACE VIEW MADV_QA_BRV_RSNID AS
SELECT OBJECTID, BASE_SEGMENT_ID, RANGE_SOURCE_NAME_ID FROM MAD_BASE_RANGE_VARIANTS
WHERE RANGE_SOURCE_NAME_ID NOT IN (SELECT DISTINCT SOURCE_NAME_ID FROM MAD_SOURCE)
OR RANGE_SOURCE_NAME_ID IS NULL
OR RANGE_SOURCE_NAME_ID = ''
ORDER BY COMMUNITY_ID_LEFT, COMMUNITY_ID_RIGHT, STREET_NAME_ID;
