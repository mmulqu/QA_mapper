# View identifies any records in MAD_BASE_RANGE_VARIANTS with an incorrect LAST_EDIT_TYPE value.  
# "Incorrect" means it is a non-null value not within the acceptable set of domain values.
# DM - 10/25/17

CREATE OR REPLACE VIEW MADV_QA_BRV_DOM_LASTEDIT AS
SELECT OBJECTID, BASE_RANGE_VARIANT_ID, LAST_EDIT_TYPE, LAST_EDIT_DATE, LAST_EDIT_BY FROM MAD_BASE_RANGE_VARIANTS BRV
WHERE LAST_EDIT_TYPE IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'LAST_EDIT_TYPE'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_BASE_RANGE_VARIANTS'
AND BRV.LAST_EDIT_TYPE = DOM.DOMAIN_VALUE)
ORDER BY LAST_EDIT_TYPE, LAST_EDIT_DATE, LAST_EDIT_BY, COMMUNITY_ID_LEFT, BASE_RANGE_VARIANT_ID, LAST_EDIT_TYPE;
