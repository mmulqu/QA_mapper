# View identifies any records in MAD_BASE_RANGE_VARIANTS where the address range values have a parity  
# that is different than what is recorded in their associated parity field (PARITY_LEFT or PARITY_RIGHT).
# Odd address range values must have a parity of "O" and even address range values must have a parity of "E"
# and mixed parity address range values must have a parity of "M".
# DM - 9/14/16
# Updated 5/11/17 to re-incorporate Boston data.
# Updated with REGEXP_SUBSTR functions to prevent TO_NUMBER errors in Oracle 19.  11/23/20 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_PARITY_EOM AS
SELECT OBJECTID, BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID, FROM_ADDRESS_LEFT, TO_ADDRESS_LEFT, PARITY_LEFT, FROM_ADDRESS_RIGHT, TO_ADDRESS_RIGHT, PARITY_RIGHT
FROM MAD_BASE_RANGE_VARIANTS
WHERE ((( MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+')),2) = 0 AND PARITY_LEFT <> 'E' ) OR (MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+')),2) = 1 AND PARITY_LEFT <> 'O' ))
          AND REGEXP_LIKE(TO_ADDRESS_LEFT,'^\d+$') 
          AND TO_ADDRESS_LEFT IS NOT NULL
          AND TO_ADDRESS_LEFT <> ' '
          AND PARITY_LEFT <> 'M')
OR    ((( MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+')),2) = 0 and PARITY_LEFT <> 'E' ) OR (MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+')),2) = 1 and PARITY_LEFT <> 'O' ))
          AND REGEXP_LIKE(FROM_ADDRESS_LEFT,'^\d+$') 
          AND FROM_ADDRESS_LEFT IS NOT NULL
          AND FROM_ADDRESS_LEFT <> ' '
          AND PARITY_LEFT <> 'M')
OR    ((( MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')),2) = 0 AND PARITY_RIGHT <> 'E' ) OR (MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')),2) = 1 AND PARITY_RIGHT <> 'O' ))
          AND REGEXP_LIKE(TO_ADDRESS_RIGHT,'^\d+$') 
          AND TO_ADDRESS_RIGHT IS NOT NULL
          AND TO_ADDRESS_RIGHT <> ' '
          AND PARITY_RIGHT <> 'M')
OR    ((( MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')),2) = 0 and PARITY_RIGHT <> 'E' ) OR (MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')),2) = 1 and PARITY_RIGHT <> 'O' ))
          AND REGEXP_LIKE(FROM_ADDRESS_RIGHT,'^\d+$') 
          AND FROM_ADDRESS_RIGHT IS NOT NULL
          AND FROM_ADDRESS_RIGHT <> ' '
          AND PARITY_RIGHT <> 'M')
OR   (((MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+')),2) = 0 AND MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+')),2) = 1 AND PARITY_LEFT <> 'M')
       OR 
      (MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_LEFT, '\d+')),2) = 1 AND MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_LEFT, '\d+')),2) = 0 AND PARITY_LEFT <> 'M'))
      AND regexp_like(FROM_ADDRESS_LEFT,'^\d+$') AND regexp_like(TO_ADDRESS_LEFT,'^\d+$') 
      AND FROM_ADDRESS_LEFT IS NOT NULL AND TO_ADDRESS_LEFT IS NOT NULL
      AND FROM_ADDRESS_LEFT <> ' ' AND TO_ADDRESS_LEFT <> ' '
-- TOGGLED TO INCLUDE BOSTON COMMUNITIES AS OF 20170511:
--      AND (COMMUNITY_ID_LEFT NOT IN
--           (SELECT COMMUNITY_ID FROM MAD_MSAG_COMMUNITY_POLYM
--            WHERE ADDRESS_TOWN_ID = 35)
--            OR 
--            COMMUNITY_ID_RIGHT NOT IN
--           (SELECT COMMUNITY_ID FROM MAD_MSAG_COMMUNITY_POLYM
--            WHERE ADDRESS_TOWN_ID = 35))
    )
OR   (((MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')),2) = 0 AND MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')),2) = 1 AND PARITY_RIGHT <> 'M')
       OR 
      (MOD(TO_NUMBER(REGEXP_SUBSTR(FROM_ADDRESS_RIGHT, '\d+')),2) = 1 AND MOD(TO_NUMBER(REGEXP_SUBSTR(TO_ADDRESS_RIGHT, '\d+')),2) = 0 AND PARITY_RIGHT <> 'M'))
      AND regexp_like(FROM_ADDRESS_RIGHT,'^\d+$') AND regexp_like(TO_ADDRESS_RIGHT,'^\d+$') 
      AND FROM_ADDRESS_RIGHT IS NOT NULL AND TO_ADDRESS_RIGHT IS NOT NULL
      AND FROM_ADDRESS_RIGHT <> ' ' AND TO_ADDRESS_RIGHT <> ' '
-- TOGGLED TO INCLUDE BOSTON COMMUNITIES AS OF 20170511:
--      AND (COMMUNITY_ID_LEFT NOT IN
--           (SELECT COMMUNITY_ID FROM MAD_MSAG_COMMUNITY_POLYM
--            WHERE ADDRESS_TOWN_ID = 35)
--            OR 
--            COMMUNITY_ID_RIGHT NOT IN
--           (SELECT COMMUNITY_ID FROM MAD_MSAG_COMMUNITY_POLYM
--            WHERE ADDRESS_TOWN_ID = 35))
    )
;
