# View identifies any records in MAD_BASE_RANGE_VARIANTS where MSAG_STNAME_VARIANT_ID should 
# be populated but isn't.
# DM - 9/14/16
# This view was deleted and rendered obsolete when the MSAG_STNAME_VARIANT_ID became obsolete.  9/27/21 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_STNMVID_MATCH AS            
SELECT OBJECTID, BASE_SEGMENT_ID, BASE_RANGE_VARIANT_ID, STREET_NAME_ID, MSAG_STNAME_VARIANT_ID
FROM MAD_BASE_RANGE_VARIANTS
WHERE MSAG_STNAME_VARIANT_ID IS NULL
AND STREET_NAME_ID IN (
   SELECT DISTINCT STREET_NAME_ID FROM MAD_STREET_NAME_VARIANTS
   WHERE SOURCE_NAME_ID = 
        (SELECT MAX(SOURCE_NAME_ID)
         FROM MAD_SOURCE
         WHERE SOURCE_TYPE = 'MSAG'))
ORDER BY STREET_NAME_ID;
