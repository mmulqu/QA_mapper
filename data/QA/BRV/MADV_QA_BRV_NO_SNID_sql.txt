# View identifies any records in MAD_BASE_RANGE_VARIANTS with an incorrect STREET_NAME_ID.  "Incorrect" means it is either
# blank, or has an ID that doesn't exist in MAD_BASE_SEGMENT_ARC.
# DM - 9/7/16

CREATE OR REPLACE VIEW MADV_QA_BRV_NO_SNID AS
SELECT * FROM MAD_BASE_RANGE_VARIANTS BRV
WHERE STREET_NAME_ID IS NULL
OR STREET_NAME_ID = ''
OR NOT EXISTS (SELECT * FROM MAD_MASTER_STREET_NAME MSTR
WHERE MSTR.STREET_NAME_ID = BRV.STREET_NAME_ID)
ORDER BY COMMUNITY_ID_LEFT, COMMUNITY_ID_RIGHT, STREET_NAME_ID;
