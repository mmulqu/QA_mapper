# View identifies any records in MAD_MASTER_STREET_NAME with a value in COMMUNITY_ID that
# doesn't match the COMMUNITY_ID of associated records in MAD_STREET_NAME_VARIANTS or
# MAD_BASE_RANGE_VARIANTS.
# DM - 10/4/16
# Re-incorporated Boston data by commenting out the exclusion statement.  10/19/17 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_COMMID_MISMATCH AS
SELECT DISTINCT MSN.STREET_NAME, MSN.STREET_NAME_ID, MSN.COMMUNITY_ID AS MSN_COMM_ID, SNV.COMMUNITY_ID AS SNV_COMM_ID,
BRV.COMMUNITY_ID_LEFT AS BRV_COMM_LEFT, BRV.COMMUNITY_ID_RIGHT AS BRV_COMM_RIGHT, BRV.COMMUNITY_DIFFERENCE
FROM MAD_MASTER_STREET_NAME MSN
JOIN MAD_STREET_NAME_VARIANTS SNV ON SNV.STREET_NAME_ID = MSN.STREET_NAME_ID
JOIN MAD_BASE_RANGE_VARIANTS BRV ON BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID
WHERE (MSN.COMMUNITY_ID != BRV.COMMUNITY_ID_LEFT
       OR SNV.COMMUNITY_ID != BRV.COMMUNITY_ID_RIGHT)
AND BRV.COMMUNITY_DIFFERENCE != 'O' -- OVERLAPS ARE KNOWN ISSUES.
--AND MSN.COMMUNITY_ID NOT IN 	-- REMOVE BOSTON BECAUSE WE CANNOT EDIT THAT DATA.
--  (37, 45, 58, 85, 92, 155, 158, 189, 271, 274, 296, 352, 381)
AND BRV.ALIAS_FLAG <> 'N'
ORDER BY MSN.COMMUNITY_ID, MSN.STREET_NAME;
