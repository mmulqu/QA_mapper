# View identifies any records in MAD_BASE_RANGE_VARIANTS with an incorrect ADDRESS_RANGE_TYPE value.  
# "Incorrect" means it is either blank, or is a non-null value not within the acceptable set
# of domain values.
# DM - 9/12/16
# Updated to include new code 'C'.  10/20/16 - DM
# Updated to include new code 'T'.  9/26/17 - DM

CREATE OR REPLACE VIEW MADV_QA_BRV_DOM_RNG AS
SELECT OBJECTID, BASE_SEGMENT_ID, ADDRESS_RANGE_TYPE FROM MAD_BASE_RANGE_VARIANTS BRV
WHERE ADDRESS_RANGE_TYPE IS NULL
OR NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'ADDRESS_RANGE_TYPE'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_BASE_RANGE_VARIANTS'
AND BRV.ADDRESS_RANGE_TYPE = DOM.DOMAIN_VALUE)
ORDER BY COMMUNITY_ID_LEFT, STREET_NAME_ID, BASE_SEGMENT_ID, ADDRESS_RANGE_TYPE;
