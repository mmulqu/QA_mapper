# View identifies any records in MAD_STREET_NAME_VARIANTS that either have 
# a STREET_NAME_ID value not found in MAD_MASTER_STREET_NAME or a NULL
# STREET_NAME_ID.
# DM - 10/7/16

CREATE OR REPLACE VIEW MADV_QA_SNV_DOM_STNMID AS
SELECT * FROM MAD_STREET_NAME_VARIANTS SNV
WHERE (NOT EXISTS (SELECT * FROM MAD_MASTER_STREET_NAME MSN
                   WHERE SNV.STREET_NAME_ID = MSN.STREET_NAME_ID)  
       OR STREET_NAME_ID IS NULL)
AND COMMUNITY_ID NOT IN (37, 45, 58, 85, 92, 155, 158, 189, 271, 274, 296, 352, 381)
ORDER BY STREET_NAME_ID, COMMUNITY_ID, STREET_NAME_BASE;
