# View identifies any records in MAD_STREET_NAME_VARIANTS where the input street name is associated
# with more than one standardized street name ID (i.e. "one-to-many" or OTM issues) where the street
# name is not a ROUTE.  Many of these will likely need to be moved from street name variants and
# master street name into the MAD_SITE_NAMES table.  Some are legitimate and can remain.

# DM - 10/11/16, Updated 4/9/2026 to reflect actual SQL in use that was not updated here at same time.

CREATE OR REPLACE VIEW MADV_QA_SNV_OTM AS
WITH OTM AS (
SELECT DISTINCT FULL_STREET_NAME_PARSED, STREET_NAME_ID, COMMUNITY_ID FROM MAD_STREET_NAME_VARIANTS SNV
WHERE EXISTS (SELECT * FROM (
                                SELECT * FROM
                                                (SELECT FULL_STREET_NAME_PARSED, COMMUNITY_ID, COUNT(FULL_STREET_NAME_PARSED) AS OTM_CNT FROM
                                                                (SELECT FULL_STREET_NAME_PARSED, STREET_NAME_ID, COMMUNITY_ID
                                                                FROM MAD_STREET_NAME_VARIANTS
                                                                GROUP BY FULL_STREET_NAME_PARSED, STREET_NAME_ID, COMMUNITY_ID
                                                                ORDER BY COMMUNITY_ID, FULL_STREET_NAME_PARSED) UNIQ_SNV
                                                GROUP BY FULL_STREET_NAME_PARSED, COMMUNITY_ID
                                                ORDER BY COMMUNITY_ID, FULL_STREET_NAME_PARSED) CNT_QRY
                                WHERE OTM_CNT > 1
                                ORDER BY COMMUNITY_ID, FULL_STREET_NAME_PARSED) OTM_QRY
                      WHERE SNV.FULL_STREET_NAME_PARSED = OTM_QRY.FULL_STREET_NAME_PARSED
                      AND SNV.COMMUNITY_ID = OTM_QRY.COMMUNITY_ID)
ORDER BY COMMUNITY_ID, FULL_STREET_NAME_PARSED, STREET_NAME_ID)
                
SELECT DISTINCT OTM.FULL_STREET_NAME_PARSED, OTM.STREET_NAME_ID, MSN.STREET_NAME, OTM.COMMUNITY_ID
FROM OTM
LEFT JOIN MAD_MASTER_STREET_NAME MSN ON OTM.STREET_NAME_ID = MSN.STREET_NAME_ID
WHERE MSN.STREET_NAME NOT LIKE 'ROUTE%'
AND OTM.FULL_STREET_NAME_PARSED NOT LIKE 'ROUTE%'
ORDER BY OTM.COMMUNITY_ID, OTM.FULL_STREET_NAME_PARSED, MSN.STREET_NAME;
