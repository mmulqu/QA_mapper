# This view identifies any address records in MAD_MASTER_ADDRESS that have an incorrect
# ADDRESS_TOWN_ID value, as determined by the geographic comm/town of the street segment
# the address is associated with.
# 1/25/22 - DM


CREATE OR REPLACE VIEW MADV_QA_MA_ADDRTOWNID_MISMATCH AS
SELECT DISTINCT MA.MASTER_ADDRESS_ID, MA.FULL_NUMBER_STANDARDIZED AS FULL_NUM, MA.ADDRESS_NUMBER, MSN.STREET_NAME, MA.STREET_NAME_ID, 
MA.ADDRESS_TOWN_ID MA_TOWN_ID, MC.ADDRESS_TOWN_ID BSA_TOWN_ID, BRV.BASE_RANGE_VARIANT_ID, BRV.FROM_ADDRESS_LEFT, BRV.TO_ADDRESS_LEFT, 
BRV.FROM_ADDRESS_RIGHT, BRV.TO_ADDRESS_RIGHT, BRV.BASE_SEGMENT_ID, BSA.COMMUNITY_ID_LEFT_ADDRESS BSA_COMM_ID_LEFT, BSA.COMMUNITY_ID_RIGHT_ADDRESS BSA_COMM_ID_RIGHT
FROM MAD_MASTER_ADDRESS MA, MAD_MASTER_STREET_NAME MSN, MAD_MSAG_COMMUNITY_POLYM MC, MAD_BASE_RANGE_VARIANTS BRV, MAD_BASE_STREET_ARC BSA
WHERE MA.STREET_NAME_ID = MSN.STREET_NAME_ID 
--AND MA.COMMUNITY_ID = MC.COMMUNITY_ID 
AND MA.STREET_NAME_ID = BRV.STREET_NAME_ID 
AND BRV.BASE_SEGMENT_ID = BSA.BASE_SEGMENT_ID
AND (MA.ADDRESS_STATUS NOT IN ('INACTIVE', 'NOT_FOUND') OR MA.ADDRESS_STATUS IS NULL)
AND (
      (NOT REGEXP_LIKE(BRV.TO_ADDRESS_LEFT, '\D')
       AND NOT REGEXP_LIKE(BRV.FROM_ADDRESS_LEFT, '\D')
       AND ( ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(BRV.FROM_ADDRESS_LEFT) AND TO_NUMBER(BRV.TO_ADDRESS_LEFT))
              OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(BRV.TO_ADDRESS_LEFT) AND TO_NUMBER(BRV.FROM_ADDRESS_LEFT)))
            AND BSA.COMMUNITY_ID_LEFT_ADDRESS = MC.COMMUNITY_ID
            AND ((MC.ADDRESS_TOWN_ID <> MA.ADDRESS_TOWN_ID AND MC.COMMUNITY_ID <> 83)
                 OR (MC.COMMUNITY_ID = 83 AND MA.ADDRESS_TOWN_ID NOT IN (19, 125, 147, 270)))
            )
       AND(
         (MOD(MA.ADDRESS_NUMBER, 2) = 0
          AND (BRV.PARITY_LEFT = 'E' OR BRV.PARITY_LEFT = ''))
          OR
         (MOD(MA.ADDRESS_NUMBER, 2) = 1
          AND (BRV.PARITY_LEFT = 'O' OR BRV.PARITY_LEFT = ''))    
          )
       )
    OR
       (NOT REGEXP_LIKE(BRV.TO_ADDRESS_RIGHT, '\D')
       AND NOT REGEXP_LIKE(BRV.FROM_ADDRESS_RIGHT, '\D')
       AND ( ((MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(BRV.FROM_ADDRESS_RIGHT) AND TO_NUMBER(BRV.TO_ADDRESS_RIGHT))
               OR (MA.ADDRESS_NUMBER BETWEEN TO_NUMBER(BRV.TO_ADDRESS_RIGHT) AND TO_NUMBER(BRV.FROM_ADDRESS_RIGHT)))
            AND BSA.COMMUNITY_ID_RIGHT_ADDRESS = MC.COMMUNITY_ID
            AND ((MC.ADDRESS_TOWN_ID <> MA.ADDRESS_TOWN_ID AND MC.COMMUNITY_ID <> 83)
                 OR (MC.COMMUNITY_ID = 83 AND MA.ADDRESS_TOWN_ID NOT IN (19, 125, 147, 270)))
            )
       AND(
         (MOD(MA.ADDRESS_NUMBER, 2) = 0
          AND (BRV.PARITY_RIGHT = 'E' OR BRV.PARITY_RIGHT = ''))
          OR
         (MOD(MA.ADDRESS_NUMBER, 2) = 1
          AND (BRV.PARITY_RIGHT = 'O' OR BRV.PARITY_RIGHT = ''))    
          )
       )      
    ) 
ORDER BY MA.ADDRESS_TOWN_ID, MSN.STREET_NAME, MA.FULL_NUMBER_STANDARDIZED;
