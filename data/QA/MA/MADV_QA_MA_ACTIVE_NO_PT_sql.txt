# SQL view to show all records in MAD_MASTER_ADDRESS that have an ADDRESS_STATUS of 'ACTIVE'
# but the ADDRESS_POINT_ID value is NULL.  An address can only be active if it links to geography.
# DM - 11/3/16
# Edited to include 'LINKED'.  6/21/19 - DM

CREATE OR REPLACE VIEW MADV_QA_MA_ACTIVE_NO_PT AS
SELECT * FROM MAD_MASTER_ADDRESS
WHERE ADDRESS_STATUS IN ('ACTIVE','LINKED')
AND ADDRESS_POINT_ID IS NULL 
ORDER BY GEOGRAPHIC_TOWN_ID, STREET_NAME_ID, FULL_NUMBER_STANDARDIZED;
