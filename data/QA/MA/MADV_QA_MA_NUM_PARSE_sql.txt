# A view that identifies any records in MAD_MASTER_ADDRESS where the parsed address number fields do not 
# represent the information in FULL_NUMBER_STANDARDIZED when concatenated, like they should.  Corrections
# will likely only be made to the parsed address number fields and not FULL_NUMBER_STANDARDIZED values
# unless it is shown that FULL_NUMBER_STANDARDIZED is wrong (i.e. has a non-standardized value).
# DM - 8/30/16

CREATE OR REPLACE VIEW MADV_QA_MA_NUM_PARSE AS
SELECT * FROM MAD_MASTER_ADDRESS
WHERE (FULL_NUMBER_STANDARDIZED <> TRIM(ADDRESS_NUMBER_PREFIX||ADDRESS_NUMBER||ADDRESS_NUMBER_SUFFIX)
AND FULL_NUMBER_STANDARDIZED <> TRIM(ADDRESS_NUMBER_PREFIX||ADDRESS_NUMBER||' '||ADDRESS_NUMBER_SUFFIX)
AND FULL_NUMBER_STANDARDIZED <> TRIM(ADDRESS_NUMBER_PREFIX||ADDRESS_NUMBER||ADDRESS_NUMBER_SUFFIX||'-'||ADDRESS_NUMBER_2_PREFIX||ADDRESS_NUMBER_2||ADDRESS_NUMBER_2_SUFFIX)
AND FULL_NUMBER_STANDARDIZED <> TRIM(ADDRESS_NUMBER_PREFIX||ADDRESS_NUMBER||' '||ADDRESS_NUMBER_SUFFIX||'-'||ADDRESS_NUMBER_2_PREFIX||ADDRESS_NUMBER_2||ADDRESS_NUMBER_2_SUFFIX)
AND FULL_NUMBER_STANDARDIZED <> TRIM(ADDRESS_NUMBER_PREFIX||ADDRESS_NUMBER||ADDRESS_NUMBER_SUFFIX||'-'||ADDRESS_NUMBER_2_PREFIX||ADDRESS_NUMBER_2||' '||ADDRESS_NUMBER_2_SUFFIX)
OR (FULL_NUMBER_STANDARDIZED IS NOT NULL AND (TRIM(ADDRESS_NUMBER_PREFIX||ADDRESS_NUMBER||ADDRESS_NUMBER_SUFFIX)) IS NULL))
;
