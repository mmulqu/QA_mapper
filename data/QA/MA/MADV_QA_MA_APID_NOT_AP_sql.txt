# SQL view to show all records in MAD_MASTER_ADDRESS that have a non-null address point ID
# but that address point ID doesn't appear in MAD_ADDRESS_POINTM or MAD_ADDRESS_POINTM_CENTROID.  (where did the point go?)
# DM - 6/30/14
# Updated to include centroid check.  8/26/16 - DM

CREATE OR REPLACE VIEW MADV_QA_MA_APID_NOT_AP AS
SELECT * FROM MAD_MASTER_ADDRESS MA
WHERE ADDRESS_POINT_ID IS NOT NULL
AND ADDRESS_POINT_ID NOT LIKE 'MULTI%'
AND NOT EXISTS (SELECT * FROM MAD_ADDRESS_POINTM AP
                WHERE AP.ADDRESS_POINT_ID = MA.ADDRESS_POINT_ID)
AND NOT EXISTS (SELECT * FROM MAD_ADDRESS_POINTM_CENTROID CNTRD
                WHERE CNTRD.CENTROID_ID = MA.ADDRESS_POINT_ID)
AND (ADDRESS_STATUS IS NULL OR ADDRESS_STATUS NOT IN ('NOT_FOUND','DELETED','INACTIVE')) ;
