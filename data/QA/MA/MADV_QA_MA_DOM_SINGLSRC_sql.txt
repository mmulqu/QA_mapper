# View identifies any records in MAD_MASTER_ADDRESS with an unacceptable SINGLE_SOURCE value.  
# "Unacceptable" means it is a non-null value not within the acceptable set of domain values.
# DM - 6/24/19

CREATE OR REPLACE VIEW MADV_QA_MA_DOM_SINGLSRC AS
SELECT * FROM MAD_MASTER_ADDRESS MA
WHERE SINGLE_SOURCE IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'SINGLE_SOURCE'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_MASTER_ADDRESS'
AND MA.SINGLE_SOURCE = DOM.DOMAIN_VALUE)
ORDER BY SINGLE_SOURCE, GEOGRAPHIC_TOWN_ID, STREET_NAME_ID, FULL_NUMBER_STANDARDIZED, UNIT;
