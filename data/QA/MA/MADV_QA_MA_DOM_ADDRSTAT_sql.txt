# View identifies any records in MAD_MASTER_ADDRESS with an unacceptable ADDRESS_STATUS value.  
# "Unacceptable" means it is NULL or a non-null value not within the acceptable set of domain values.
# DM - 10/26/17

CREATE OR REPLACE VIEW MADV_QA_MA_DOM_ADDRSTAT AS
SELECT * FROM MAD_MASTER_ADDRESS MA
WHERE ADDRESS_STATUS IS NULL
OR NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'ADDRESS_STATUS'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_MASTER_ADDRESS'
AND MA.ADDRESS_STATUS = DOM.DOMAIN_VALUE)
ORDER BY ADDRESS_STATUS, GEOGRAPHIC_TOWN_ID, STREET_NAME_ID, FULL_NUMBER_STANDARDIZED, UNIT;
