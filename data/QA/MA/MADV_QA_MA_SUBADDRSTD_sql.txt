# View identifies any records in MAD_MASTER_ADDRESS where either...
# 1. SUBADDRESS_STANDARDIZED is NULL and one or more of the subaddress fields BUILDING_NAME, WING, FLOOR, UNIT,
# UNIT_TYPE, ROOM are not NULL
# OR
# 2. SUBADDRESS_STANDARDIZED is not NULL but one or more of the subaddress fields does not have its value(s) 
# represented in SUBADDRESS_STANDARDIZED
# OR
# 3. SUBADDRESS_STANDARDIZED is not NULL and all of the subaddress fields are NULL
# DM - 6/21/19

CREATE OR REPLACE VIEW MADV_QA_MA_SUBADDRSTD AS
SELECT OBJECTID, MASTER_ADDRESS_ID, FULL_NUMBER_STANDARDIZED, REL_LOC, STREET_NAME_ID, SUBADDRESS_STANDARDIZED,
BUILDING_NAME, WING, FLOOR, UNIT, UNIT_TYPE, ROOM, ADDRESS_POINT_ID 
FROM MAD_MASTER_ADDRESS MA
WHERE (SUBADDRESS_STANDARDIZED IS NULL
AND     (BUILDING_NAME IS NOT NULL
         OR WING IS NOT NULL
         OR FLOOR IS NOT NULL
         OR UNIT IS NOT NULL
         OR UNIT_TYPE IS NOT NULL
         OR ROOM IS NOT NULL))
OR (SUBADDRESS_STANDARDIZED IS NOT NULL
AND (    (BUILDING_NAME IS NOT NULL AND NOT REGEXP_LIKE (SUBADDRESS_STANDARDIZED, BUILDING_NAME||'( |,|$)'))
      OR (WING IS NOT NULL AND NOT REGEXP_LIKE (SUBADDRESS_STANDARDIZED, WING||'( |,|$)'))
      OR (FLOOR IS NOT NULL AND NOT REGEXP_LIKE (SUBADDRESS_STANDARDIZED, FLOOR||'( |,|$)'))
      OR (UNIT IS NOT NULL AND NOT REGEXP_LIKE (SUBADDRESS_STANDARDIZED, UNIT||'( |,|$)'))
      OR (UNIT_TYPE IS NOT NULL AND NOT REGEXP_LIKE (SUBADDRESS_STANDARDIZED, UNIT_TYPE||'( |,|$)'))
      OR (ROOM IS NOT NULL AND NOT REGEXP_LIKE (SUBADDRESS_STANDARDIZED, ROOM||'( |,|$)'))
    ))
OR (SUBADDRESS_STANDARDIZED IS NOT NULL
    AND BUILDING_NAME IS NULL
    AND WING IS NULL
    AND FLOOR IS NULL
    AND UNIT IS NULL
    AND UNIT_TYPE IS NULL
    AND ROOM IS NULL)
ORDER BY GEOGRAPHIC_TOWN_ID, STREET_NAME_ID, FULL_NUMBER_STANDARDIZED, BUILDING_NAME, WING, FLOOR, UNIT, ROOM;
