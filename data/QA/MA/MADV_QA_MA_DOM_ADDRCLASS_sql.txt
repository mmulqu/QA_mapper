# View identifies any records in MAD_MASTER_ADDRESS with an unacceptable ADDRESS_CLASS value.  
# "Unacceptable" means it is a non-null value not within the acceptable set of domain values.
# DM - 6/21/19

CREATE OR REPLACE VIEW MADV_QA_MA_DOM_ADDRCLASS AS
SELECT * FROM MAD_MASTER_ADDRESS MA
WHERE ADDRESS_CLASS IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'ADDRESS_CLASS'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_MASTER_ADDRESS'
AND MA.ADDRESS_CLASS = DOM.DOMAIN_VALUE)
ORDER BY ADDRESS_CLASS, GEOGRAPHIC_TOWN_ID, STREET_NAME_ID, FULL_NUMBER_STANDARDIZED, UNIT;
