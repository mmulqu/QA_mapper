# SQL view to show all records in MAD_MASTER_ADDRESS that don't have any source variant records
# in MAD_ADDRESS_VARIANTS based on the MASTER_ADDRESS_ID relationship.  Records without any source
# address variants either need to have a source record identified or created for them, or are candidates
# for deletion.
# DM - 6/20/23

CREATE OR REPLACE VIEW MADV_QA_MA_NO_VARIANTS AS
SELECT MA.MASTER_ADDRESS_ID, MA.ADDRESS_POINT_ID, MA.FULL_NUMBER_STANDARDIZED, MA.REL_LOC, SN.STREET_NAME, MA.STREET_NAME_ID,
MA.BUILDING_NAME, MA.WING, MA.FLOOR, MA.UNIT, MA.UNIT_TYPE, MA.ROOM, MA.OTHER_LOCATION, STN.SITE_NAME, MA.SITE_NAME_ID, 
MA.NON_BUILDING_STRUCTURE_NAME, MA.COMMUNITY_ID, MA.ADDRESS_TOWN_ID, MA.GEOGRAPHIC_TOWN_ID, MA.ADDRESS_STATUS, MA.DATE_CREATED, 
MA.LAST_EDIT_DATE, MA.LAST_EDIT_BY, MA.LAST_EDIT_COMMENTS, MA.SINGLE_SOURCE, MA.MA_UUID, MA.ID_OF_PREVIOUS_ADDRESS, 
MA.ID_OF_REPLACEMENT_ADDRESS
FROM MAD_MASTER_ADDRESS MA
LEFT JOIN MAD_MASTER_STREET_NAME SN ON MA.STREET_NAME_ID = SN.STREET_NAME_ID
LEFT JOIN MAD_SITE_NAMES STN ON MA.SITE_NAME_ID = STN.SITE_NAME_ID
WHERE NOT EXISTS (SELECT * FROM MAD_ADDRESS_VARIANTS AV
WHERE MA.MASTER_ADDRESS_ID = AV.MASTER_ADDRESS_ID)
ORDER BY STREET_NAME, FULL_NUMBER_STANDARDIZED, REL_LOC, OTHER_LOCATION, SITE_NAME, BUILDING_NAME, 
FLOOR, UNIT, ROOM, NON_BUILDING_STRUCTURE_NAME;

CREATE OR REPLACE VIEW MADV_QA_MA_NO_VARIANTS AS
SELECT * FROM MAD_MASTER_ADDRESS MA
WHERE NOT EXISTS (SELECT * FROM MAD_ADDRESS_VARIANTS AV
WHERE MA.MASTER_ADDRESS_ID = AV.MASTER_ADDRESS_ID)
ORDER BY GEOGRAPHIC_TOWN_ID, STREET_NAME_ID, FULL_NUMBER_STANDARDIZED, BUILDING_NAME, WING, 
FLOOR, UNIT, ROOM, OTHER_LOCATION;
