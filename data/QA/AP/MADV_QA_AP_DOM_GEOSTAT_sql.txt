# View identifies any features in MAD_ADDRESS_POINTM with an unacceptable GEOGRAPHIC_EDIT_STATUS value.  
# "Unacceptable" means it is a non-null value not within the acceptable set of domain values.
# DM - 10/23/17

CREATE OR REPLACE VIEW MADV_QA_AP_DOM_GEOSTAT AS
SELECT * FROM MAD_ADDRESS_POINTM AP
WHERE GEOGRAPHIC_EDIT_STATUS IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'GEOGRAPHIC_EDIT_STATUS'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_ADDRESS_POINTM'
AND AP.GEOGRAPHIC_EDIT_STATUS = DOM.DOMAIN_VALUE)
ORDER BY GEOGRAPHIC_EDIT_STATUS, GEOGRAPHIC_TOWN_ID;
