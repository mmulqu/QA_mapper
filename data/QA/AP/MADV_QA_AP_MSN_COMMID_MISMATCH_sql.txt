# This view shows any address points where the COMMUNITY_ID is different than the COMMUNITY_ID
# of the street name in MAD_MASTER_STREET_NAME that is populated in the master address(es) linked
# to the address point, and those two different communities are not adjacent, indicating a significant
# error.  The fix may be to update an erroneous COMMUNITY_ID assigned to the address point, or a fix
# to the STREET_NAME_ID of the associated master address(es).
# 1/31/25 - DM (via EA and PG)

CREATE OR REPLACE VIEW MADV_QA_AP_MSN_COMMID_MISMATCH AS
  WITH A AS
    (
    SELECT MC1.COMMUNITY_ID COMM_1, MC2.COMMUNITY_ID COMM_2
    FROM MAD_MSAG_COMMUNITY_POLYM MC1, MAD_MSAG_COMMUNITY_POLYM MC2
    WHERE SDE.ST_INTERSECTS(MC1.SHAPE, MC2.SHAPE) = 1
    AND MC1.COMMUNITY_ID <> MC2.COMMUNITY_ID
    )
SELECT AP.SHAPE, MA.MASTER_ADDRESS_ID, AP.COMMUNITY_ID AP_COMMUNITY_ID, MA.FULL_NUMBER_STANDARDIZED, MSN.STREET_NAME, MA.BUILDING_NAME, MA.UNIT, AP.ADDRESS_POINT_ID, MSN.COMMUNITY_ID MSN_COMMUNITY_ID
FROM MAD_MASTER_ADDRESS MA
JOIN MAD_ADDRESS_POINTM AP
ON MA.ADDRESS_POINT_ID = AP.ADDRESS_POINT_ID
JOIN MAD_MASTER_STREET_NAME MSN
ON MA.STREET_NAME_ID = MSN.STREET_NAME_ID
WHERE AP.COMMUNITY_ID <> MSN.COMMUNITY_ID
AND MA.ADDRESS_STATUS NOT IN ('INACTIVE', 'NOT_FOUND')
AND MSN.COMMUNITY_ID NOT IN
    (
    SELECT COMM_2
    FROM A
    WHERE COMM_1 = AP.COMMUNITY_ID
    )
AND NOT (MA.GEOGRAPHIC_TOWN_ID = 314 AND MA.ADDRESS_TOWN_ID = 308)
AND NOT (MA.GEOGRAPHIC_TOWN_ID = 308 AND MA.ADDRESS_TOWN_ID = 314)
AND NOT (MA.GEOGRAPHIC_TOWN_ID = 137 AND MA.ADDRESS_TOWN_ID = 329)
AND NOT (MA.GEOGRAPHIC_TOWN_ID = 325 AND MA.ADDRESS_TOWN_ID = 329)
AND 
    (
    MA.LAST_EDIT_COMMENTS NOT LIKE '%54%'
    OR
    MA.LAST_EDIT_COMMENTS IS NULL
    )
ORDER BY AP_COMMUNITY_ID, MSN.STREET_NAME, MA.FULL_NUMBER_STANDARDIZED, MA.BUILDING_NAME, MA.UNIT;
