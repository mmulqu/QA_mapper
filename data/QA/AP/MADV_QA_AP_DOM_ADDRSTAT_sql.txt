# View identifies any features in MAD_ADDRESS_POINTM with an unacceptable ADDRESS_STATUS value.  
# "Unacceptable" means it is a non-null value not within the acceptable set of domain values.
# DM - 10/10/17

CREATE OR REPLACE VIEW MADV_QA_AP_DOM_ADDRSTAT AS
SELECT * FROM MAD_ADDRESS_POINTM AP
WHERE ADDRESS_STATUS IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'ADDRESS_STATUS'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_ADDRESS_POINTM'
AND AP.ADDRESS_STATUS = DOM.DOMAIN_VALUE);
