# View identifies any features in MAD_ADDRESS_POINTM with an unacceptable STRUCTURE_STATUS value.  
# "Unacceptable" means it is either a null value or not within the acceptable set of domain values.
# DM - 10/24/17

CREATE OR REPLACE VIEW MADV_QA_AP_DOM_STRCTSTAT AS
SELECT * FROM MAD_ADDRESS_POINTM AP
WHERE STRUCTURE_STATUS IS NULL
OR NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'STRUCTURE_STATUS'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_ADDRESS_POINTM'
AND AP.STRUCTURE_STATUS = DOM.DOMAIN_VALUE)
OR (POINT_TYPE IN ('BC', 'BMP', 'DBMP') AND STRUCTURE_STATUS = 'N')
OR (POINT_TYPE = 'ABC' AND STRUCTURE_STATUS = 'N' AND (GEOGRAPHIC_EDIT_STATUS IS NULL OR GEOGRAPHIC_EDIT_STATUS <> 'PLANNED'))
ORDER BY POINT_TYPE, STRUCTURE_STATUS, GEOGRAPHIC_TOWN_ID;
