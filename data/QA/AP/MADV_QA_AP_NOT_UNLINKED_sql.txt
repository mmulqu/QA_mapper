# View shows any address points in MAD_ADDRESS_POINTM that have an ADDRESS_STATUS of 'UNLINKED', but are
# associated with a master address record with an ADDRESS_STATUS that is something other than 'NOT_FOUND',
# 'INACTIVE', or 'UNLINKED'.  The address point likely needs to have its ADDRESS_STATUS value updated to
# be 'LINKED' or NULL/empty.
# DM - 10/10/17

CREATE OR REPLACE VIEW MADV_QA_AP_NOT_UNLINKED AS
SELECT * FROM MAD_MASTER_ADDRESS MA
WHERE EXISTS (SELECT * FROM MAD_ADDRESS_POINTM AP
WHERE ADDRESS_STATUS = 'UNLINKED' 
AND AP.ADDRESS_POINT_ID = MA.ADDRESS_POINT_ID)
AND ADDRESS_STATUS NOT IN ('NOT_FOUND', 'INACTIVE', 'UNLINKED')
ORDER BY GEOGRAPHIC_TOWN_ID, STREET_NAME_ID, FULL_NUMBER_STANDARDIZED;
