# View shows any address points in MAD_ADDRESS_POINTM that is missing an associated centroid
# in MAD_ADDRESS_POINTM_CENTROID.  Every address point should have a simple point centroid, 
# especially if it links to an address so it appears in the 911 data products.
# DM - 3/22/16

CREATE OR REPLACE VIEW MADV_QA_AP_NO_CNTRD AS
SELECT * FROM MAD_ADDRESS_POINTM AP
WHERE NOT EXISTS (SELECT * FROM MAD_ADDRESS_POINTM_CENTROID APC
WHERE AP.ADDRESS_POINT_ID = APC.CENTROID_ID)
AND NOT EXISTS (SELECT * FROM MAD_ADDRESS_POINTM_DISSOLVED DISS
WHERE AP.ADDRESS_POINT_ID = DISS.ORIGINAL_ADDRESS_POINT_ID)
ORDER BY GEOGRAPHIC_TOWN_ID, ADDRESS_POINT_ID;
