# This view shows any address point records where COMMUNITY_ID AND GEOGRAPHIC_TOWN_ID
# are not in sync.
# 10/27/16 - DM

CREATE OR REPLACE VIEW MADV_QA_AP_TOWNCOMM_MISMATCH AS
SELECT * FROM MAD_ADDRESS_POINTM AP
WHERE EXISTS (SELECT * FROM MAD_MSAG_COMMUNITY_POLYM MSAG
              WHERE AP.COMMUNITY_ID = MSAG.COMMUNITY_ID
              AND (AP.GEOGRAPHIC_TOWN_ID <> MSAG.ADDRESS_TOWN_ID
                   OR AP.GEOGRAPHIC_TOWN_ID IS NULL AND MSAG.ADDRESS_TOWN_ID IS NOT NULL
                   OR AP.GEOGRAPHIC_TOWN_ID IS NOT NULL AND MSAG.ADDRESS_TOWN_ID IS NULL)
              AND AP.COMMUNITY_ID <> 83)
OR (COMMUNITY_ID > 1000 AND GEOGRAPHIC_TOWN_ID IS NOT NULL)
OR (COMMUNITY_ID < 1000 AND GEOGRAPHIC_TOWN_ID IS NULL);
