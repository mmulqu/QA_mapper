# View identifies any records in MAD_EMERGENCY_SERVICE_ZONES (ESZ's) where there are
# non-unique combinations of ESN and ROUTE_URI.

# DM - 10/11/16

CREATE OR REPLACE VIEW MADV_QA_ESZ_ROUTE_URI_CNT AS
SELECT ESN, ROUTE_URI, CNT
FROM (SELECT ESN, ROUTE_URI, COUNT (ROUTE_URI) OVER (PARTITION BY ESN ORDER BY ESN) AS CNT
      FROM (SELECT DISTINCT ESN, ROUTE_URI FROM mad_emergency_service_zones ))
WHERE CNT > 1;
