# View identifies any emergency service zones that are erroneously associated with multiple MSAG communities.
# IT IS INTENDED TO FIND PLACES WHERE A ESN WAS UPDATED INCORRECTLY SO THAT MORE THAN ONE COMM IS ASSOCIATED WITH A PARTICULAR ESN.
# EXCEPTIONS OCCUR WHEN TOWNS THAT HAVE MULTIPLE MSAG COMMUNITIES OVERLAP INTO ADJACENT TOWNS. IN THESE CASES THE NAME OF THE ORIGINAL 
# COMMUNITY, AS IT RELATES TO THE MSAG, IS KEPT.
# EXCEPTIONS ARE REMOVED IN THE QUERY BELOW. THESE EXCEPTIONS MAY HAVE TO BE UPDATED.
# DM - 10/11/16

CREATE OR REPLACE VIEW MADV_QA_ESZ_MULTICOMM AS
SELECT ESN, MSAG_COMM, CNT
FROM (SELECT ESN, MSAG_COMM, COUNT (MSAG_COMM) OVER (PARTITION BY ESN ORDER BY ESN) AS CNT
      FROM (SELECT DISTINCT ESN, MSAG_COMM FROM MAD_EMERGENCY_SERVICE_ZONES ))
WHERE CNT > 1
AND ESN || ' ' || MSAG_COMM NOT IN
('138 EAST BOSTON', '138 HYDE PARK', '138 WEST ROXBURY', '138 CHARLESTOWN', '138 BRIGHTON', '138 BOSTON',
'320 FLORENCE', '320 NORTHAMPTON', '232 NORTHAMPTON', '232 HATFIELD')
ORDER BY ESN;
