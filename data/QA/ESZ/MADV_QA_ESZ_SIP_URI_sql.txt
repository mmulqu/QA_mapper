# View identifies any records in MAD_EMERGENCY_SERVICE_ZONES (ESZ's) that are associated
# with incorrect SIP URI values.
# PREVIOUSLY CREATED A TABLE IN SDE CALLED MAD.MAD_PSAP_SIP_URIS THAT 'FLATTENS' THE OFFICIALS TABLE DELIVERED BY MA911 
# CALLED SIP URI_Policy_20150430.xlsx FOUND HERE: V:\Projects\SETB_NT\MAD\ESZ\Wireless_ESNs\DATA\DDTI_WIRELESS_LUT. 
# THE SIP URI_Policy TABLE HAS DISTINCT SIP URI VALUES AND HAS MULTIPLE ESN VALUES IN THE ESN FIELD SEPARATED BY COMMAS. 
# THE TABLE MAD_PSAP_SIP_URIS ADDS ONE OR MORE ROWS FOR EACH ESN FIELD THAT CONTAINS MULTIPLE ESN VALUES SO THAT ESN NOW 
# CONTAINS ONLY ONE ESN VALUE.
# DM - 10/11/16

CREATE OR REPLACE VIEW MADV_QA_ESZ_SIP_URI AS
WITH POLY_PSAP
AS (SELECT URI1.PSAP, ESZ1.ESN
    FROM MAD_EMERGENCY_SERVICE_ZONES ESZ1, MAD_PSAP_SIP_URIS URI1
    WHERE URI1.NOMINAL_SIP_URI = ESZ1.ROUTE_URI)
SELECT DISTINCT ESZ.ES_UNIQUE_ID AS POLY_ID, ESZ.MSAG_COMM AS POLY_COMM, ESZ.ROUTE_URI AS POLY_URI, POLY_PSAP.PSAP AS POLY_PSAP, URI.PSAP AS TABLE_PSAP, URI.NOMINAL_SIP_URI AS TABLE_URI
FROM MAD_EMERGENCY_SERVICE_ZONES ESZ, MAD_PSAP_SIP_URIS URI, POLY_PSAP POLY_PSAP
WHERE ESZ.ESN = URI.ESN
AND URI.ESN = POLY_PSAP.ESN
AND ESZ.ROUTE_URI != URI.NOMINAL_SIP_URI;
