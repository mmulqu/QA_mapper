# View identifies any records in MAD_EMERGENCY_SERVICE_ZONES polygons where the value
# in PRIMARY_PS (the PSAP name) is different than the PSAP name that appears in
# MAD_PSAP_SIP_URIS, in the field 'PSAP'.  
# THE PSAP VALUE IN MAD_PSAP_SIP_URIS NEEDS TO UPDATED, WHERE APPROPRIATE, SO THAT IT MATCHES MAD_EMERGENCY_SERVICE_ZONES.

# DM - 10/12/16
# FIPS_STCO and EMERGENCY_SERVICE_ZONE_ID deleted from data and removed from selection statement here.  10/6/17 - DM
# EMERGENCY_SERVICE_ZONE_ID replaced with ES_UNIQUE_ID.  10/6/17 - DM

CREATE OR REPLACE VIEW MADV_QA_ESZ_PSAP_NAME AS
SELECT ESZ.OBJECTID, ESZ.PRIMARY_PS, UPPER(URI.PSAP) AS URI_PSAP, ESZ.TOWN, ESZ.TOWN_ID, ESZ.MSAG_COMM, ESZ.ESN, ESZ.PSAP_ID, ESZ.ES_UNIQUE_ID
FROM MAD_EMERGENCY_SERVICE_ZONES ESZ, MAD_PSAP_SIP_URIS URI
WHERE ESZ.ESN = URI.ESN
AND ESZ.PRIMARY_PS != UPPER(URI.PSAP);
