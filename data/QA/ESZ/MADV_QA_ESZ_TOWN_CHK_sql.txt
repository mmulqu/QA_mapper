# View identifies any records in MAD_EMERGENCY_SERVICE_ZONES (ESZ's) that have a TOWN and TOWN_ID
# combination not found in GISDATA.TOWNSSURVEY_POLYM.
# NOTE:  THIS VIEW REQUIRES THAT GISDATA.TOWNSSURVEY_POLYM HAS GRANTED SPECIAL READ PERMISSIONS TO BE USED IN A VIEW.
# DM - 6/4/17
# Updated 6/23/20 to account for neighboring states -- DM

CREATE OR REPLACE VIEW MADV_QA_ESZ_TOWN_CHK AS
SELECT * FROM (
SELECT DISTINCT TOWN, TOWN_ID FROM MAD_EMERGENCY_SERVICE_ZONES WHERE TOWN != 'OCEAN' AND TOWN_ID IS NOT NULL
MINUS 
SELECT DISTINCT TOWN, TOWN_ID FROM GISDATA.TOWNSSURVEY_POLYM)
WHERE TOWN NOT LIKE '%, CT'
AND TOWN NOT LIKE '%, NH'
AND TOWN NOT LIKE '%, NY'
AND TOWN NOT LIKE '%, RI'
AND TOWN NOT LIKE '%, VT';
