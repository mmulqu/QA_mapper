# View identifies any records in MAD_MASTER_STREET_NAME that have a value in COMMUNITY_ID
# that is outside the accepted domain.
# DM - 10/3/16

CREATE OR REPLACE VIEW MADV_QA_MSN_COMMID AS
SELECT * FROM MAD_MASTER_STREET_NAME
WHERE (COMMUNITY_ID NOT IN (SELECT DISTINCT COMMUNITY_ID FROM MAD_MSAG_COMMUNITY_POLYM)
       OR COMMUNITY_ID IS NULL
       OR COMMUNITY_ID = ''
       OR REGEXP_LIKE(COMMUNITY_ID, ' +'))  
AND street_name_id NOT IN (-1); -- FDC LITE PLACEHOLDER
