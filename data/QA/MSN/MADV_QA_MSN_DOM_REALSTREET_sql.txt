# View identifies any records in MAD_MASTER_STREET_NAME that have a value in REAL_STREET
# that is outside the accepted domain.
# DM - 10/4/16

CREATE OR REPLACE VIEW MADV_QA_MSN_DOM_REALSTREET AS
SELECT OBJECTID, STREET_NAME, STREET_NAME_ID, COMMUNITY_ID, REPLACE(NVL(REGEXP_REPLACE(REAL_STREET, ' +', 'SPACE'), 'NULL'), '', 'NULL STRING') REALSTREET
FROM MAD_MASTER_STREET_NAME MSN
WHERE REAL_STREET IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'REAL_STREET'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_MASTER_STREET_NAME'
AND MSN.REAL_STREET = DOM.DOMAIN_VALUE)
ORDER BY COMMUNITY_ID, STREET_NAME, STREET_NAME_ID;
