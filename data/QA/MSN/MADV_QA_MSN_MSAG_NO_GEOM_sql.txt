# Here is the pure SQL without the python formatting.

CREATE OR REPLACE VIEW MADV_QA_MSN_MSAG_NO_GEOM AS
SELECT DISTINCT SNV.OBJECTID AS SNV_OBJECTID, SNV.SOURCE_NAME_ID, SNV.COMMUNITY_ID, SNV.COMMUNITY_NAME, SNV.FULL_STREET_NAME_INPUT, SNV.STREET_NAME_ID, SNV.STREET_NAME_VARIANT_ID, SNV.NOTES, SNV.REAL_STREET,
IMNB.STATUS, IMNB.NOTES AS IMNB_NOTES, IMNB.RES_PART
FROM (SELECT DISTINCT SNV.OBJECTID, SNV.SOURCE_NAME_ID, SNV.COMMUNITY_ID, M.COMMUNITY_NAME, SNV.FULL_STREET_NAME_INPUT, SNV.STREET_NAME_ID, SNV.STREET_NAME_VARIANT_ID, MSN.NOTES, MSN.REAL_STREET, SNV.ID_IN_ORIG_SOURCE
      FROM MAD_STREET_NAME_VARIANTS SNV, MAD_MSAG_COMMUNITY_POLYM M, MAD_MASTER_STREET_NAME MSN
      WHERE SNV.COMMUNITY_ID = M.COMMUNITY_ID
      AND SNV.STREET_NAME_ID = MSN.STREET_NAME_ID
      AND SNV.STREET_NAME_ID IN
	      (SELECT STREET_NAME_ID FROM
		      (SELECT DISTINCT STREET_NAME_ID		-- REMOVED COMMUNIY_ID TO CATCH RECORDS WHERE THE STREET NAME WAS COLLAPSED ACROSS THE TOWN BOUNDARY.
    		   FROM MAD_STREET_NAME_VARIANTS 		-- DURING THE COLLAPSE THE MSAG RECORDS FROM BOTH TOWNS GET ASSOCIATED WITH THE SAME SNID.
							                              -- BY REMOVING THE COMMUNITY_ID ALL ASSOCIATED RECORDS WILL WILL GET MATCHED.
    		   WHERE SOURCE_NAME_ID = '2848'
    		   MINUS
    		   SELECT STREET_NAME_ID
    		   FROM MAD_BASE_RANGE_VARIANTS))
   AND NOT EXISTS
 	   (SELECT BRV.STREET_NAME_ID
 	    FROM MAD_BASE_RANGE_VARIANTS BRV
 	    WHERE BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID)
   AND SNV.SOURCE_NAME_ID = '2848'
   AND FULL_STREET_NAME_INPUT NOT LIKE 'NTAS %'
   AND FULL_STREET_NAME_INPUT NOT LIKE 'CETC %'
   AND FULL_STREET_NAME_INPUT NOT IN
  	 (SELECT MSAG_STREET FROM SETB.MSAG_20160623
  	  WHERE ESN LIKE '9%' 
  	  OR ESN LIKE '7%' 
  	  OR ESN LIKE '6%' 
  	  OR ESN = '535')
   AND (MSN.REAL_STREET IS NULL OR MSN.REAL_STREET != 'M')) SNV -- REMOVED 'M's FROM THE VIEW
LEFT OUTER JOIN SETB.IMNB_TESTCOPY IMNB ON  IMNB.STREET_NAME_ID = SNV.STREET_NAME_ID
ORDER BY SNV.COMMUNITY_NAME, SNV.FULL_STREET_NAME_INPUT;
