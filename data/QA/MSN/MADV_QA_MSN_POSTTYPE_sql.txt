# View identifies any records in MAD_MASTER_STREET_NAME that incorrectly have a  
# NULL value in POST_TYPE.
# DM - 10/3/16

CREATE OR REPLACE VIEW MADV_QA_MSN_POSTTYPE AS
SELECT * FROM MAD_MASTER_STREET_NAME
WHERE (STR_NAME_BASE LIKE '%STREET%'
       AND STREET_NAME NOT LIKE '%RAMP%'
       AND REAL_STREET NOT IN ('A', 'E'))
AND POST_TYPE IS NULL
ORDER BY STREET_NAME;
