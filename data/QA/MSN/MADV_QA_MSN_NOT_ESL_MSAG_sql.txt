# Here is the pure SQL without the python formatting.

CREATE OR REPLACE VIEW MADV_QA_MSN_NOT_ESL_MSAG AS
SELECT DISTINCT SNV.OBJECTID AS SNV_OBJECTID, SNV.SOURCE_NAME_ID, SNV.COMMUNITY_ID, SNV.COMMUNITY_NAME, SNV.FULL_STREET_NAME_INPUT, SNV.STREET_NAME_ID, SNV.STREET_NAME_VARIANT_ID, SNV.NOTES,
IMNB.STATUS, IMNB.NOTES AS IMNB_NOTES, IMNB.RES_PART
FROM 
	(SELECT DISTINCT SNV.OBJECTID, SNV.SOURCE_NAME_ID, SNV.COMMUNITY_ID, MC.COMMUNITY_NAME, SNV.FULL_STREET_NAME_INPUT, SNV.STREET_NAME_ID, SNV.STREET_NAME_VARIANT_ID, MSN.NOTES, SNV.ID_IN_ORIG_SOURCE
	 FROM MAD_STREET_NAME_VARIANTS SNV
	 JOIN MAD_MASTER_STREET_NAME MSN ON SNV.STREET_NAME_ID = MSN.STREET_NAME_ID
	 JOIN MAD_MSAG_COMMUNITY_POLYM MC ON SNV.COMMUNITY_ID = MC.COMMUNITY_ID	
	 JOIN SETB.ESL_20160623 ESL ON SNV.FULL_STREET_NAME_INPUT = ESL.FULL_STR AND MC.COMMUNITY_NAME = ESL.COMMUNITY
	 WHERE SNV.STREET_NAME_ID IN
		  (SELECT STREET_NAME_ID FROM
			  (SELECT DISTINCT STREET_NAME_ID
         FROM MAD_STREET_NAME_VARIANTS 
         WHERE SOURCE_NAME_ID = '2706'
         MINUS
         SELECT STREET_NAME_ID
         FROM MAD_BASE_RANGE_VARIANTS))
	 AND NOT EXISTS
		 (SELECT BRV.STREET_NAME_ID
		  FROM MAD_BASE_RANGE_VARIANTS BRV
		  WHERE BRV.STREET_NAME_VARIANT_ID = SNV.STREET_NAME_VARIANT_ID)
	 AND SNV.SOURCE_NAME_ID = '2706'
	 AND FULL_STREET_NAME_INPUT NOT LIKE 'NTAS %'
	 AND FULL_STREET_NAME_INPUT NOT LIKE 'CETC %'
	 AND FULL_STREET_NAME_INPUT NOT IN
  		(SELECT MSAG_STREET FROM SETB.MSAG_20160623
  		 WHERE ESN LIKE '9%' OR ESN LIKE '7%' OR ESN LIKE '6%' OR ESN = '535')
	-- AND (MSN.REAL_STREET IS NULL OR REAL_STREET = 'Y' OR REAL_STREET = 'V')
  AND (MSN.REAL_STREET != 'M')) SNV
LEFT OUTER JOIN SETB.IMNB_TESTCOPY IMNB ON  IMNB.STREET_NAME_ID = SNV.STREET_NAME_ID
ORDER BY SNV.COMMUNITY_NAME, SNV.FULL_STREET_NAME_INPUT;
