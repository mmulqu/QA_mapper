# View identifies any records in MAD_MASTER_STREET_NAME that are duplicates 
# based on STREET_NAME.  Duplicate street name records need to either be collapsed
# or if one record is incorrect it needs to be edited to reflect the correct,
# non-duplicated street name.
# DM - 10/3/16

CREATE OR REPLACE VIEW MADV_QA_MSN_DUPE_STRTNM AS
SELECT A.STREET_NAME, A.STREET_NAME_ID, B.STREET_NAME_ID DUPE_STREETNAME_ID
FROM MAD_MASTER_STREET_NAME A, MAD_MASTER_STREET_NAME B
WHERE A.COMMUNITY_ID = B.COMMUNITY_ID
AND A.STREET_NAME = B.STREET_NAME
AND A.STREET_NAME_ID <> B.STREET_NAME_ID
AND A.NEIGHBORHOOD IS NULL
AND B.NEIGHBORHOOD IS NULL
ORDER BY A.STREET_NAME;
