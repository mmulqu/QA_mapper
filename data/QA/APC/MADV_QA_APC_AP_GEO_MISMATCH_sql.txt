# View shows any address point centroids in MAD_ADDRESS_POINTM_CENTROID that have a LOC_ID and/or GEOGRAPHIC_TOWN_ID
# value that doesn't match the LOC_ID and/or GEOGRAPHIC_TOWN_ID of their associated address points in
# MAD_ADDRESS_POINTM.
# DM - 12/8/16

CREATE OR REPLACE VIEW MAD.MADV_QA_APC_AP_GEO_MISMATCH AS 
SELECT APC.OBJECTID, APC.CENTROID_ID, APC.LOC_ID AS APC_LOC_ID, AP.LOC_ID AS AP_LOC_ID, APC.GEOGRAPHIC_TOWN_ID AS APC_TOWN_ID, AP.GEOGRAPHIC_TOWN_ID AS AP_TOWN_ID, AP.POINT_TYPE
FROM MAD_ADDRESS_POINTM_CENTROID APC, MAD_ADDRESS_POINTM AP
WHERE (AP.ADDRESS_POINT_ID = APC.CENTROID_ID
AND AP.POINT_TYPE <> 'DBMP'
AND ((APC.LOC_ID IS NULL AND AP.LOC_ID IS NOT NULL AND AP.LOC_ID <> 'MULTIPLE')
      OR (APC.LOC_ID <> AP.LOC_ID AND AP.LOC_ID <> 'MULTIPLE')))
OR (AP.ADDRESS_POINT_ID = APC.CENTROID_ID
AND APC.GEOGRAPHIC_TOWN_ID <> AP.GEOGRAPHIC_TOWN_ID)
ORDER BY APC_TOWN_ID, AP_TOWN_ID, AP_LOC_ID;
