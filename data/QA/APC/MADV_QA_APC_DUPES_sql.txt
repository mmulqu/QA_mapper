# View shows any address point centroids in MAD_ADDRESS_POINTM_CENTROID that exist as unintentional duplicates.
# (non-unique CENTROID_ID)  Every centroid must have a unique CENTROID_ID value.  Duplicates need to be deleted 
# if obsolete.
# DM - 8/29/16

CREATE OR REPLACE VIEW MAD.MADV_QA_APC_DUPES AS 
SELECT * FROM MAD_ADDRESS_POINTM_CENTROID APC
WHERE EXISTS
   (SELECT * FROM
      (SELECT DISTINCT CENTROID_ID, COUNT (CENTROID_ID) AS CNTRDCOUNT FROM MAD_ADDRESS_POINTM_CENTROID
       GROUP BY CENTROID_ID) QRY
    WHERE CNTRDCOUNT > 1
    AND APC.CENTROID_ID = QRY.CENTROID_ID)
ORDER BY GEOGRAPHIC_TOWN_ID, CENTROID_ID;
