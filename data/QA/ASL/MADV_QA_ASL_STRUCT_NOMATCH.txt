# A view that identifies any records in MAD_ADDPT_STRUCT_LUT where the STRUCTURE_ID doesn't represent
# any structure polygon in MAD_STRUCTURES_POLY.
# DM - 6/19/17

CREATE OR REPLACE VIEW MADV_QA_ASL_STRUCT_NOMATCH AS
SELECT * FROM MAD_ADDPT_STRUCT_LUT LUT
WHERE NOT EXISTS (SELECT * FROM MAD_STRUCTURES_POLY ST
WHERE LUT.STRUCTURE_ID = ST.STRUCTURE_ID)
AND STRUCTURE_ID IS NOT NULL
ORDER BY STRUCTURE_TOWN_ID, STRUCTURE_ID;
