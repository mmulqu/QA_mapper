# A view that identifies any records in MAD_ADDPT_STRUCT_LUT that are duplicates of each other
# and therefore are unnecessary. The duplicates should be deleted, leaving only a single copy
# of each record.
# DM - 10/3/17

CREATE OR REPLACE VIEW MADV_QA_ASL_DUPES AS
SELECT DISTINCT OBJ_ID AS OBJECTID, ADDRESS_POINT_ID, LOC_ID, STRUCTURE_ID, STRUCTURE_TOWN_ID FROM (
SELECT MAX(ASL.OBJECTID) OVER (PARTITION BY ASL.ADDRESS_POINT_ID, ASL.LOC_ID, ASL.STRUCTURE_ID, ASL.STRUCTURE_TOWN_ID) AS OBJ_ID,
ASL.ADDRESS_POINT_ID, ASL.LOC_ID, ASL.STRUCTURE_ID, ASL.STRUCTURE_TOWN_ID
FROM MAD_ADDPT_STRUCT_LUT ASL, 
(SELECT LUT1.OBJECTID, LUT1.ADDRESS_POINT_ID, LUT1.LOC_ID, LUT1.STRUCTURE_ID, LUT1.STRUCTURE_TOWN_ID
FROM MAD_ADDPT_STRUCT_LUT LUT1, MAD_ADDPT_STRUCT_LUT LUT2
WHERE LUT1.OBJECTID <> LUT2.OBJECTID
AND (LUT1.ADDRESS_POINT_ID = LUT2.ADDRESS_POINT_ID OR (LUT1.ADDRESS_POINT_ID IS NULL AND LUT2.ADDRESS_POINT_ID IS NULL))
AND (LUT1.LOC_ID = LUT2.LOC_ID OR (LUT1.LOC_ID IS NULL AND LUT2.LOC_ID IS NULL))
AND (LUT1.STRUCTURE_ID = LUT2.STRUCTURE_ID OR (LUT1.STRUCTURE_ID IS NULL AND LUT2.STRUCTURE_ID IS NULL))
AND (LUT1.STRUCTURE_TOWN_ID = LUT2.STRUCTURE_TOWN_ID OR (LUT1.STRUCTURE_TOWN_ID IS NULL AND LUT2.STRUCTURE_TOWN_ID IS NULL))
ORDER BY STRUCTURE_TOWN_ID, STRUCTURE_ID, OBJECTID) QRY
WHERE ASL.OBJECTID = QRY.OBJECTID)
ORDER BY STRUCTURE_TOWN_ID, ADDRESS_POINT_ID, OBJECTID;
