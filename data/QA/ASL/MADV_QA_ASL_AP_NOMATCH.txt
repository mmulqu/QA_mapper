# A view that identifies any records in MAD_ADDPT_STRUCT_LUT where the ADDRESS_POINT_ID doesn't represent
# any address point in MAD_ADDRESS_POINTM.
# DM - 6/19/17

CREATE OR REPLACE VIEW MADV_QA_ASL_AP_NOMATCH AS 
SELECT * FROM MAD_ADDPT_STRUCT_LUT LUT
WHERE NOT EXISTS (SELECT * FROM MAD_ADDRESS_POINTM AP
WHERE LUT.ADDRESS_POINT_ID = AP.ADDRESS_POINT_ID)
ORDER BY STRUCTURE_TOWN_ID, ADDRESS_POINT_ID, STRUCTURE_ID;
