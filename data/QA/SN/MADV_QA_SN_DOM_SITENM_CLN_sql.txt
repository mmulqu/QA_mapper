# View identifies any records in MAD_SITE_NAMES that have a value in CLN_FLAG
# that is outside the accepted domain.
# DM - 8/5/23

CREATE OR REPLACE VIEW MADV_QA_SN_DOM_SITENM_CLN AS
SELECT * FROM MAD_SITE_NAMES SN
WHERE CLN_FLAG IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'CLN_FLAG'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_SITE_NAMES'
AND SN.CLN_FLAG = DOM.DOMAIN_VALUE)
ORDER BY CLN_FLAG, GEOGRAPHIC_TOWN_ID, SITE_NAME;
