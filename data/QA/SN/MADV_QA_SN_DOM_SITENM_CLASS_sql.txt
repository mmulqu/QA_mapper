# View identifies any records in MAD_SITE_NAMES that have a value in SITENAME_CLASS
# that is outside the accepted domain.
# DM - 8/5/23

CREATE OR REPLACE VIEW MADV_QA_SN_DOM_SITENM_CLASS AS
SELECT * FROM MAD_SITE_NAMES SN
WHERE SITENAME_CLASS IS NOT NULL
AND NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'SITENAME_CLASS'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_SITE_NAMES'
AND SN.SITENAME_CLASS = DOM.DOMAIN_VALUE)
ORDER BY SITENAME_CLASS, GEOGRAPHIC_TOWN_ID, SITE_NAME;
