# View identifies any records in MAD_SITE_NAMES that have a value in SITENAME_SOURCETYPE
# that is outside the accepted domain.
# DM - 10/30/17

CREATE OR REPLACE VIEW MADV_QA_SN_DOM_SITENM_SRCTYPE AS
SELECT * FROM MAD_SITE_NAMES SN
WHERE SITENAME_SOURCETYPE IS NULL
OR NOT EXISTS (SELECT * FROM MADMETA_DOMAINS DOM
WHERE DOM.DOMAIN_FIELD = 'SITENAME_SOURCETYPE'
AND DOM.ENTITY_WITH_DOMAIN = 'MAD_SITE_NAMES'
AND SN.SITENAME_SOURCETYPE = DOM.DOMAIN_VALUE)
ORDER BY SITENAME_SOURCETYPE, GEOGRAPHIC_TOWN_ID, SITE_NAME;
